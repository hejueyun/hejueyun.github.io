12.11 并行课程：按层出队，层数就是最少学期数 了解

了解即可：拓扑排序的进阶变形。点击展开

原题回顾

课程表 207/210（10.12）：有 n 门课和先修关系，问能不能修完、按什么顺序修。

变化点

LeetCode 1136：每个学期可以同时修任意多门课，只要它的先修课都在之前的学期修完了。问最少几个学期能修完全部课；有环修不完就返回 −1。本节课程编号用 0..n−1。[1]

为什么这样改：一层就是一个学期

第 1 学期能修的，正是入度为 0 的全部课程；修完它们后新解锁的课（入度减到 0 的）要等第 2 学期。所以把 Kahn 改成"按层出队"：每一轮先记下此刻队列里有几门课 cnt，把这 cnt 门全部取出来算作同一学期；它们解锁的课进队列，留到下一轮。轮数就是学期数。

每学期都把能修的全修了，不会比任何安排更慢，所以层数就是最少学期数。换个角度看，层数等于图中最长依赖链上的课程数：链上的课只能一学期修一门。

代码

int minSemesters(int n, const vector<pair<int, int>>& edges) {        // 课程 0..n-1，边 (u,v)：先 u 后 v
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int semesters = 0, taken = 0;
    while (!q.empty()) {
        ++semesters;                                                   // 开始新的一学期
        int cnt = q.size();                                            // 这学期修的 = 此刻队列里的全部
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            ++taken;
            for (int v : g[u]) if (--indeg[v] == 0) q.push(v);         // 新解锁的课下学期修
        }
    }
    return taken == n ? semesters : -1;                                // 有环：永远修不完
}
def min_semesters(n, edges):                            # 课程 0..n-1，边 (u,v)：先 u 后 v
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    semesters = taken = 0
    while q:
        semesters += 1                                  # 开始新的一学期
        for _ in range(len(q)):                         # range 只在开头算一次 len(q)：此刻队列里的全部
            u = q.popleft()
            taken += 1
            for v in g[u]:
                indeg[v] -= 1
                if indeg[v] == 0:                       # 新解锁的课下学期修
                    q.append(v)
    return semesters if taken == n else -1              # 有环：永远修不完

练一练：排出每学期的课表

题面： 不只要学期数，还要输出每学期修哪几门（同一学期内按出队顺序）。上面的例子应输出 [[0,1],[2,4],[3],[5]]；有环返回空。

vector<vector<int>> semesterPlan(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    vector<vector<int>> plan;
    int taken = 0;
    while (!q.empty()) {
        int cnt = q.size();
        plan.push_back({});                                            // 改动 1：每层开一个新列表
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            plan.back().push_back(u);                                  // 改动 2：记下这学期修了谁
            ++taken;
            for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
        }
    }
    if (taken < n) return {};
    return plan;
}
def semester_plan(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    plan = []
    taken = 0
    while q:
        plan.append([])                                 # 改动 1：每层开一个新列表
        for _ in range(len(q)):
            u = q.popleft()
            plan[-1].append(u)                          # 改动 2：记下这学期修了谁
            taken += 1
            for v in g[u]:
                indeg[v] -= 1
                if indeg[v] == 0:
                    q.append(v)
    return plan if taken == n else []

与例题的区别： 每层开始时 plan.push_back({})；出队时把课程放进 plan.back()；学期数就是 plan.size()。

单选 · 理解检查 1　5 门课，边 0→1、0→2、1→3、2→3、3→4。最少几个学期？如果再加一条边 4→0 呢？

A. 4 个学期；加边后返回 −1

B. 5 个学期；加边后 5 个

C. 3 个学期；加边后返回 −1

D. 4 个学期；加边后 4 个

查看答案与每个选项的解析

A · 正确：分层是 {0}、{1,2}、{3}、{4}。加 4→0 后 0→1→3→4→0 成环，没有入度为 0 的点，taken = 0。

B · 错误：1 和 2 可以在同一学期修，5 门课只要 4 层。

C · 错误：3 要等 1、2 修完，4 要等 3 修完，最长链 0→1→3→4 有 4 门课。

D · 错误：有环时环上的课永远修不完，按约定返回 −1。

练习 12.11-1【基础】数据流水线最少跑几轮

题面：数据平台有 n 个任务，编号 1..n。rel 中的 (a, b) 表示任务 a 的输出是任务 b 的输入，b 只能在 a 所在轮次之后的某一轮运行。每一轮可以并行运行任意多个任务。写 int minRounds(int n, rel) 返回跑完全部任务的最少轮数，有循环依赖返回 −1。

样例：n = 5，rel = [(1,3),(2,3),(3,4),(3,5),(2,5)]，输出 3：第 1 轮跑 1、2，第 2 轮跑 3，第 3 轮跑 4、5。n = 3，rel = [(1,2),(2,3),(3,1)]，输出 −1。

提示

一轮就是一学期，题目就是 minSemesters，只差编号从 1 开始：数组开 n + 1 个位置，找起点的循环从 1 到 n，下标 0 不用。判环仍然比较"跑过的任务数"和 n。

参考答案
int minRounds(int n, const vector<pair<int, int>>& rel) {   // 任务编号 1..n
    vector<vector<int>> g(n + 1);
    vector<int> indeg(n + 1, 0);
    for (auto [a, b] : rel) { g[a].push_back(b); ++indeg[b]; }
    queue<int> q;
    for (int i = 1; i <= n; ++i) if (indeg[i] == 0) q.push(i);   // 下标 0 不用
    int rounds = 0, ran = 0;
    while (!q.empty()) {
        ++rounds;
        int cnt = q.size();
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            ++ran;
            for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
        }
    }
    return ran == n ? rounds : -1;
}
def min_rounds(n, rel):                                 # 任务编号 1..n
    g = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    for a, b in rel:
        g[a].append(b)
        indeg[b] += 1
    q = deque(i for i in range(1, n + 1) if indeg[i] == 0)   # 下标 0 不用
    rounds = ran = 0
    while q:
        rounds += 1
        for _ in range(len(q)):
            u = q.popleft()
            ran += 1
            for v in g[u]:
                indeg[v] -= 1
                if indeg[v] == 0:
                    q.append(v)
    return rounds if ran == n else -1

练习 12.11-2【变形】每门课最早第几学期修

题面：n 门课（编号 0..n−1）、先修边表 edges。每学期能修任意多门，先修课必须在之前的学期修完。写 vector<int> earliestSemester(int n, edges)，返回每门课最早能在第几学期修（从 1 数）；有环返回空数组。

样例：12.11 手工模拟的 6 门课（边 0→2、1→2、2→3、1→4、4→5、3→5），输出 [1,1,2,3,2,4]。5 号课要等 3 号（第 3 学期）和 4 号（第 2 学期），取较晚的，排在第 4 学期。

提示

可以按层出队，把层号记给出队的课。更直接的写法不分层：在沿边更新时写 sem[v] = max(sem[v], sem[u] + 1)，没有先修的课 sem = 1。这和 12.12 的"等最慢的前驱"同一个结构：v 要等最晚修完的那门先修课，再往后一学期。u 出队时它的先修都已出队并更新过它，sem[u] 已经最终确定。

参考答案
vector<int> earliestSemester(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    vector<int> sem(n, 1);                                   // 没有先修的课第 1 学期修
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int taken = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++taken;
        for (int v : g[u]) {
            sem[v] = max(sem[v], sem[u] + 1);                // 等最晚的那门先修
            if (--indeg[v] == 0) q.push(v);
        }
    }
    return taken == n ? sem : vector<int>{};
}
def earliest_semester(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    sem = [1] * n                                       # 没有先修的课第 1 学期修
    q = deque(i for i in range(n) if indeg[i] == 0)
    taken = 0
    while q:
        u = q.popleft()
        taken += 1
        for v in g[u]:
            sem[v] = max(sem[v], sem[u] + 1)            # 等最晚的那门先修
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return sem if taken == n else []

练习 12.11-3【变形】哪个学期课最多

题面：按"每学期把所有能修的课都修掉"来排课。写 int maxPerSemester(int n, edges)，返回课最多的那个学期修了几门；有环返回 −1。

样例：12.11 手工模拟的 6 门课，每学期分别修 [0,1]、[2,4]、[3]、[5]，输出 2。n = 4 没有边，四门课都在第 1 学期，输出 4。

提示

按层出队时，每层开始先记下的 cnt 就是这学期修的门数，对它取最大值即可，其余沿用 minSemesters。这个最大值属于"能修就修"这一种排法；如果允许把不急的课往后挪，同一学期的门数可以更少，那是另一个问题。

参考答案
int maxPerSemester(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int best = 0, taken = 0;
    while (!q.empty()) {
        int cnt = q.size();
        best = max(best, cnt);                               // 这一层的门数
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            ++taken;
            for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
        }
    }
    return taken == n ? best : -1;
}
def max_per_semester(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    best = taken = 0
    while q:
        cnt = len(q)
        best = max(best, cnt)                           # 这一层的门数
        for _ in range(cnt):
            u = q.popleft()
            taken += 1
            for v in g[u]:
                indeg[v] -= 1

练习 12.11-4【变形】已经修过一部分课

题面：n 门课、先修边表 edges，done 里的课已经修过（不论它们的先修是否修过）。写 int remainingSemesters(int n, edges, done)，返回修完其余课程最少还要几学期；全部修过返回 0；其余课程之间有环返回 −1。

提示

已修的课相当于从图里删掉：指向它的边没有意义（它不用再修），从它出发的边已经满足（它已经修完）。所以只在建图时改：两端都没修的边才加入。起点只找没修的课，判环时拿出队数和"没修的课数"比，而不是和 n 比。如果建图时保留了 (1,4) 这类指向已修课的边，4 的入度会被减到 0，已经修过的 4 又被排进某个学期，出队数也会多于没修的课数。

int remainingSemesters(int n, const vector<pair<int, int>>& edges, const vector<int>& done) {
    vector<bool> isDone(n, false);
    for (int c : done) isDone[c] = true;
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges)
        if (!isDone[u] && !isDone[v]) { g[u].push_back(v); ++indeg[v]; }   // 只留两端都没修的边
    queue<int> q;
    int left = 0;                                            // 没修的课数
    for (int i = 0; i < n; ++i)
        if (!isDone[i]) { ++left; if (indeg[i] == 0) q.push(i); }
    int semesters = 0, taken = 0;
    while (!q.empty()) {
        ++semesters;
        int cnt = q.size();
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            ++taken;
            for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
        }
    }
    return taken == left ? semesters : -1;

def remaining_semesters(n, edges, done):

        is_done[c] = True

            g[u].append(v)
            indeg[v] += 1

练习 12.11-5【综合】有的先修可以同一学期修

题面：n 门课，rel 中每项 (u, v, w)：w = 1 表示 u 必须在 v 之前的学期修完；w = 0 表示 u 可以和 v 在同一学期修（实验课和理论课这种），但不能晚于 v。不看 w 时整张图若有环，返回 −1。写 int minSemestersCoreq(int n, rel)，返回最少学期数。

样例：n = 4，rel = [(0,1,1),(1,2,0),(0,3,0),(3,2,1)]，输出 2：0、3 在第 1 学期，1、2 在第 2 学期。n = 3，rel = [(0,1,0),(1,2,0)]，输出 1。n = 2，rel = [(0,1,0),(1,0,0)]，输出 −1。

提示

按层出队在这里会多算：w = 0 的边允许 v 和 u 同一学期，可 v 仍会排到 u 的下一层。改成在沿边更新时传递学期号：sem[v] = max(sem[v], sem[u] + w)，所有课初值为 1。这和 12.12 的 start[v] = max(start[v], finish[u]) 是同一种写法，"耗时"换成了边权 w，所以普通队列就行，不用分层。u 出队时它的前驱都已出队，sem[u] 已经确定。答案取所有 sem 的最大值；出队数少于 n 返回 −1。

参考答案
int minSemestersCoreq(int n, const vector<array<int, 3>>& rel) {   // {u, v, w}
    vector<vector<pair<int, int>>> g(n);                    // u -> {v, w}
    vector<int> indeg(n, 0);
    for (const auto& r : rel) { g[r[0]].push_back({r[1], r[2]}); ++indeg[r[1]]; }
    vector<int> sem(n, 1);
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);

        int u = q.front(); q.pop();
        ++taken;

}

易错点：

  • 写成 for (int i = 0; i < q.size(); ++i)：q.size() 每轮都在变，会把下一学期的课混进这一学期。
  • 只返回层数、不判断 taken：有环时会返回一个看起来正常的数字。
  • 题目如果额外规定"每学期最多修 m 门"，问题就变难了，按层贪心不再保证最优（这是另一道题 1494，需要状态压缩）。

小结： 最少学期数等于按层进行 Kahn 算法的层数，也等于最长依赖链上的课程数；每层开始时先记下队列长度再出队；修完的课程数小于 n 时返回 −1。

原题回顾

上一节每门课都恰好占一个学期，答案是"层数"。

变化点

LeetCode 2050：第 v 门课要连续学 time[v] 个月，可以同时学任意多门，开始学 v 之前它的先修课必须都学完。问学完全部课程最少要几个月。图保证无环。[2]

为什么这样改：等最慢的前驱

一门课最早什么时候能开工？它的所有先修课都完成的时刻，也就是先修课完成时刻的最大值；没有先修课的从 0 开工。完成时刻 = 开工时刻 + 耗时。全部课程完成时刻的最大值就是答案：

start[v] = max(finish[u])，u 取遍 v 的先修；finish[v] = start[v] + time[v]

要算 start[v]，必须先知道它所有先修的 finish，而拓扑序恰好保证"轮到 v 时，它的先修都已经处理完"。所以在骨架的位置 ⑤ 顺手更新 start[v] = max(start[v], finish[u])。这和 16.17 的 AOE 网求"事件最早发生时间"是同一个计算，只是这里耗时写在点上、16.17 写在边上。

vector<int> criticalPath(int n, const vector<pair<int, int>>& edges, const vector<int>& time) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    vector<int> start(n, 0), from(n, -1);                              // from[v]：决定 v 开工时刻的前驱
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int last = 0, best = -1;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        int finish = start[u] + time[u];
        if (finish > best) { best = finish; last = u; }                // 最晚完成的课是路径终点
        for (int v : g[u]) {
            if (finish > start[v]) { start[v] = finish; from[v] = u; } // 记下"是谁拖住了 v"
            if (--indeg[v] == 0) q.push(v);
        }
    }
    vector<int> path;
    for (int v = last; v != -1; v = from[v]) path.push_back(v);        // 顺着 from 往回走
    reverse(path.begin(), path.end());
    return path;
}
def critical_path(n, edges, time):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    start, frm = [0] * n, [-1] * n                      # frm[v]：决定 v 开工时刻的前驱（from 是关键字）
    q = deque(i for i in range(n) if indeg[i] == 0)
    last, best = 0, -1
    while q:
        u = q.popleft()
        finish = start[u] + time[u]
        if finish > best:                               # 最晚完成的课是路径终点
            best, last = finish, u
        for v in g[u]:
            if finish > start[v]:                       # 记下"是谁拖住了 v"
                start[v], frm[v] = finish, u
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    path = []
    v = last
    while v != -1:                                      # 顺着 frm 往回走
        path.append(v)
        v = frm[v]
    return path[::-1]

练习 12.12-2【变形】每道工序最多能拖多久

题面：沿用上题的输入。一道工序的"机动时间"是：它的开工时刻最多能比最早开工时刻推迟多少分钟，而不让全部完成的时刻变晚。写 vector<int> slackTimes(int n, deps, cost) 返回每道工序的机动时间。机动时间为 0 的工序就是关键工序。

样例：上题的厨房，输出 [0,5,0,27,0]。总时间 38；装盘最晚第 30 分钟开工，所以炒菜最晚第 30 分钟完成、第 10 分钟开工，和最早开工时刻相同，机动 0；切肉最晚第 10 分钟完成、第 5 分钟开工，机动 5；烧水最晚第 30 分钟完成、第 27 分钟开工，机动 27。

提示

先正着按拓扑序算最早开工时刻 es 和总时间 T，同时把出队顺序记下来。再倒着处理：最晚完成时刻 lf[v] 是 v 所有后继的最晚开工时刻中的最小值，没有后继的工序 lf = T；最晚开工 ls[v] = lf[v] − cost[v]。算 v 时它的后继必须已经算好，把正向的拓扑序倒过来走正好满足。机动时间 = ls − es。这就是 16.17 AOE 网里"最早开始"和"最迟开始"两个量。

参考答案
vector<int> slackTimes(int n, const vector<pair<int, int>>& deps, const vector<int>& cost) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [a, b] : deps) { g[a].push_back(b); ++indeg[b]; }
    vector<int> es(n, 0), order;                             // es：最早开工时刻
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int total = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        total = max(total, es[u] + cost[u]);
        for (int v : g[u]) {
            es[v] = max(es[v], es[u] + cost[u]);
            if (--indeg[v] == 0) q.push(v);
        }
    }
    vector<int> ls(n), slack(n);                             // ls：最晚开工时刻
    for (int i = n - 1; i >= 0; --i) {                       // 倒着走：后继先算好
        int u = order[i];
        int lf = total;                                      // 没有后继：最晚在总时间完成
        for (int v : g[u]) lf = min(lf, ls[v]);
        ls[u] = lf - cost[u];
        slack[u] = ls[u] - es[u];
    }
    return slack;
}
def slack_times(n, deps, cost):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for a, b in deps:
        g[a].append(b)
        indeg[b] += 1
    es, order = [0] * n, []                             # es：最早开工时刻
    q = deque(i for i in range(n) if indeg[i] == 0)
    total = 0
    while q:
        u = q.popleft()
        order.append(u)
        total = max(total, es[u] + cost[u])
        for v in g[u]:
            es[v] = max(es[v], es[u] + cost[u])
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    ls, slack = [0] * n, [0] * n                        # ls：最晚开工时刻
    for u in reversed(order):                           # 倒着走：后继先算好
        lf = total                                      # 没有后继：最晚在总时间完成
        for v in g[u]:
            lf = min(lf, ls[v])
        ls[u] = lf - cost[u]
        slack[u] = ls[u] - es[u]
    return slack

练习 12.12-3【综合】有最早开工时刻，还可能有环

题面：n 道工序（n ≤ 10⁵），cost[i] 和 release[i] 都在 0..10⁹ 之间：工序 i 要 cost[i] 分钟，且不能早于时刻 release[i] 开工（原料那时才到）。deps 中 (a, b) 表示 a 做完才能开始 b，录入可能有误而形成环。写 long long finishWithRelease(int n, deps, cost, release)，返回全部完成的最早时刻；有环返回 −1。

样例：n = 3，deps = [(0,1),(1,2)]，cost = [2,3,1]，release = [0,10,0]，输出 14：工序 1 的前驱第 2 分钟就完成了，但它要等到第 10 分钟才能开工。同一张图 cost = [10⁹, 10⁹, 10⁹]、release 全为 0，输出 3 × 10⁹。deps = [(0,1),(1,0)] 时输出 −1。

提示

三处改动。开工时刻的初值从 0 改成 release[v]，再在沿边更新时和前驱完成时刻取最大值，于是两条约束都满足后才开工。三道 10⁹ 分钟的工序串起来是 3 × 10⁹，超过 int 的上限约 2.1 × 10⁹，start、finish、答案全部用 long long。题目不再保证无环，所以要数出队个数，少于 n 返回 −1。

参考答案
long long finishWithRelease(int n, const vector<pair<int, int>>& deps,
                            const vector<int>& cost, const vector<int>& release) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [a, b] : deps) { g[a].push_back(b); ++indeg[b]; }
    vector<long long> start(release.begin(), release.end());   // 初值：原料到达时刻
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    long long ans = 0;
    int done = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++done;
        long long finish = start[u] + cost[u];               // long long：串起来会超过 int
        ans = max(ans, finish);
        for (int v : g[u]) {
            start[v] = max(start[v], finish);
            if (--indeg[v] == 0) q.push(v);
        }
    }
    return done == n ? ans : -1;                             // 有环：有工序永远开不了工
}
def finish_with_release(n, deps, cost, release):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for a, b in deps:
        g[a].append(b)
        indeg[b] += 1
    start = list(release)                               # 初值：原料到达时刻
    q = deque(i for i in range(n) if indeg[i] == 0)
    ans = done = 0
    while q:
        u = q.popleft()
        done += 1
        finish = start[u] + cost[u]                     # Python 整数不会溢出
        ans = max(ans, finish)
        for v in g[u]:
            start[v] = max(start[v], finish)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return ans if done == n else -1                     # 有环：有工序永远开不了工

易错点：

  • 把 max 写成 +：start[v] = start[v] + finish[u] 会把并行的先修当成串行累加。
  • 答案取最后出队那门课的完成时刻：最后出队的未必最晚完成，要取所有 finish 的最大值。
  • 多条关键路径耗时相同时，finish > start[v] 用严格大于只保留先遇到的那条；题目要求全部输出时得另外处理。

小结： 课程带有耗时时，start[v] 等于所有前驱 finish 的最大值，finish = start + time，答案是 finish 的最大值；记录每个点由哪个前驱确定，就能回溯出关键路径。

12.13 火星词典：从相邻单词里找出字母的先后 了解

了解即可：拓扑排序的进阶变形。点击展开

原题回顾

拓扑排序的输入通常直接给出边："先修 b 才能修 a"。

变化点

LeetCode 269：外星语言也用小写字母，但字母顺序未知。给出一个按外星字母顺序排好序的单词表，求一个与之相符的字母顺序；单词表自相矛盾时返回空串。这里边没有直接给出，要自己从单词里找出来，属于改位置 ①。为了让答案唯一、便于测试，本节在多个答案中取字典序最小的那个（用 12.10 的小根堆）。[3]

为什么这样改：只看相邻两个单词的第一处不同

字典序的规则是：从左往右比，第一处不同的字母决定先后，后面的字母不再起作用。所以：

  • 相邻两个单词 a、b（a 在前），找到第一处不同的位置 j，就得到一条边 a[j] → b[j]（a[j] 排在 b[j] 前）。
  • 只比相邻的两个就够：排好序的表里，第 1 个和第 3 个的关系可以由 1-2、2-3 两对推出来。
  • 如果没有不同的位置，说明一个是另一个的前缀：短的在前（"ab" 在 "abc" 前）是合法的，只是没有信息；长的在前（"abc" 在 "ab" 前）在任何字母顺序下都不成立，直接返回空串。

找完边后做拓扑排序；有环说明约束互相矛盾，同样返回空串。参与排序的点是所有出现过的字母，没出现在任何边上的字母也要输出。

代码

string alienOrder(const vector<string>& words) {
    vector<vector<int>> g(26);
    vector<int> indeg(26, 0);
    vector<bool> seen(26, false);
    for (const string& w : words) for (char c : w) seen[c - 'a'] = true;
    for (size_t i = 0; i + 1 < words.size(); ++i) {
        const string& a = words[i];
        const string& b = words[i + 1];
        size_t j = 0;
        while (j < a.size() && j < b.size() && a[j] == b[j]) ++j;      // 找第一处不同
        if (j == min(a.size(), b.size())) {                             // 一个是另一个的前缀
            if (a.size() > b.size()) return "";                         // "abc" 排在 "ab" 前：非法
            continue;                                                   // "ab" 在 "abc" 前：合法但没信息
        }
        g[a[j] - 'a'].push_back(b[j] - 'a');                            // a[j] 排在 b[j] 前
        ++indeg[b[j] - 'a'];
    }
    priority_queue<int, vector<int>, greater<int>> pq;                  // 小根堆：答案唯一，便于测试
    int total = 0;
    for (int c = 0; c < 26; ++c)
        if (seen[c]) { ++total; if (indeg[c] == 0) pq.push(c); }
    string order;
    while (!pq.empty()) {
        int u = pq.top(); pq.pop();
        order += char('a' + u);
        for (int v : g[u]) if (--indeg[v] == 0) pq.push(v);
    }
    return (int)order.size() == total ? order : "";                     // 有环：返回空串
}
def alien_order(words):
    g = [[] for _ in range(26)]
    indeg = [0] * 26
    seen = [False] * 26
    for w in words:
        for c in w:
            seen[ord(c) - 97] = True                    # ord('a') == 97
    for a, b in zip(words, words[1:]):                  # 相邻两个单词
        j = 0
        while j < len(a) and j < len(b) and a[j] == b[j]:   # 找第一处不同
            j += 1
        if j == min(len(a), len(b)):                    # 一个是另一个的前缀
            if len(a) > len(b):                         # "abc" 排在 "ab" 前：非法
                return ""
            continue                                    # "ab" 在 "abc" 前：合法但没信息
        g[ord(a[j]) - 97].append(ord(b[j]) - 97)        # a[j] 排在 b[j] 前
        indeg[ord(b[j]) - 97] += 1
    pq = [c for c in range(26) if seen[c] and indeg[c] == 0]   # 小根堆：答案唯一，便于测试
    heapq.heapify(pq)
    total = sum(seen)
    order = []
    while pq:

        order.append(chr(97 + u))
        for v in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                heapq.heappush(pq, v)
    return "".join(order) if len(order) == total else ""   # 有环：返回空串

练习 12.13-3【变形】两个字母谁在前

题面：给按外星字母顺序排好序、且不矛盾的单词表 words，以及两个出现过的不同字母 x、y。写 int compareLetters(words, char x, char y)：所有相符的字母顺序里 x 都在 y 前，返回 1；y 都在 x 前，返回 −1；两种都有，返回 0。

提示

"x 一定在 y 前"等价于字母图里有一条从 x 到 y 的路径。有路径时，路径上每条边都是必须满足的先后，连起来就是 x 在 y 前。没有路径时，可以构造一个 y 在 x 前的合法顺序：先把 y 和 y 的全部祖先按拓扑序排好（x 不是 y 的祖先，所以不在其中），再排剩下的。建图沿用 alienOrder，然后从 x 做一次 BFS 看能否到 y，再从 y 做一次看能否到 x。图无环，所以两个方向不会同时可达。

bool letterReach(const vector<vector<int>>& g, int s, int t) {   // 字母图里 s 能否走到 t
    vector<bool> vis(26, false);
    queue<int> q;
    q.push(s);
    vis[s] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        if (u == t) return true;
        for (int v : g[u]) if (!vis[v]) { vis[v] = true; q.push(v); }
    }
    return false;
}

        size_t j = 0;

        if (j < a.size() && j < b.size()) g[a[j] - 'a'].push_back(b[j] - 'a');   // 题目保证不矛盾
    }
    if (letterReach(g, x - 'a', y - 'a')) return 1;          // 有路径：x 一定在 y 前
    if (letterReach(g, y - 'a', x - 'a')) return -1;
    return 0;                                                // 两个方向都不通：先后不确定
}
def letter_reach(g, s, t):                              # 字母图里 s 能否走到 t
    vis = [False] * 26
    q = deque([s])
    vis[s] = True
    while q:
        u = q.popleft()
        if u == t:
            return True
        for v in g[u]:
            if not vis[v]:
                vis[v] = True
                q.append(v)
    return False

练习 12.13-4【变形】按外星字母顺序给单词排序

题面：给外星字母顺序 order（26 个字母的一个排列）和单词表 words，写 vector<string> sortAlien(words, order)，按外星字典序从小到大排序后返回。比较规则：第一处不同的字母按 order 中的先后比；没有不同时短的在前。

样例：order = "hlabcdefgijkmnopqrstuvwxyz"，words = ["leet","hello","lee","hi"]，输出 ["hello","hi","lee","leet"]。h 在 order 里排第 0，l 排第 1，所以 h 开头的在前；"hello" 和 "hi" 第一处不同是 e 与 i，e 排第 6、i 排第 9；"lee" 是 "leet" 的前缀，排在前面。

提示

把每个字母换成它在 order 里的名次，单词就变成一个整数序列；外星字典序恰好是这些序列的普通字典序，"短的前缀在前"也一并满足。C++ 的 vector<int> 和 Python 的 list 本身就按字典序比较，所以把 {名次序列, 单词} 放在一起排序即可，不必手写比较函数。

参考答案
vector<string> sortAlien(const vector<string>& words, const string& order) {
    int rk[26] = {0};
    for (int i = 0; i < (int)order.size(); ++i) rk[order[i] - 'a'] = i;   // 字母 -> 名次
    vector<pair<vector<int>, string>> keyed;
    for (const string& w : words) {
        vector<int> key;
        for (char c : w) key.push_back(rk[c - 'a']);         // 单词 -> 名次序列
        keyed.push_back({key, w});
    }
    sort(keyed.begin(), keyed.end());                        // vector 本身按字典序比较
    vector<string> out;
    for (const auto& kw : keyed) out.push_back(kw.second);
    return out;
}
def sort_alien(words, order):
    rk = {ch: i for i, ch in enumerate(order)}          # 字母 -> 名次
    return sorted(words, key=lambda w: [rk[c] for c in w])   # list 本身按字典序比较

练习 12.13-5【综合】从哪个单词开始出现矛盾

题面：单词表 words（n ≤ 10⁴ 个单词，可以有空串）是一个个录入的，录入员可能在某一步出了错。写 int firstConflict(words)：返回最小的 j，使前 j + 1 个单词 words[0..j] 已经自相矛盾（存在长单词排在自己前缀之前，或字母约束成环）；整张表都不矛盾时返回 −1。

样例：["ab","ba","bc","ca","cb","a"] 输出 5：前 5 个单词给出 a 在 b 前、a 在 c 前、b 在 c 前，都相容；第 6 个 "a" 排在 "cb" 后面，给出 c 在 a 前，和 a 在 c 前成环。["abc","ab"] 输出 1。["z","x","z"] 输出 2。["","",""] 输出 −1。

提示

前 j + 1 个单词矛盾，再加单词只会多出约束，仍然矛盾；所以"前缀是否矛盾"随 j 单调：前一段都是"不矛盾"，后一段都是"矛盾"，可以二分找第一个"矛盾"。判断一段单词是否矛盾，用火星词典的建图加 Kahn，单独写成返回 bool 的函数。不能拿 alienOrder 是否返回空串来判断：全是空串的单词表没有字母，合法顺序本来就是空串。二分的 check 每次 O(总字符数)，共 O(总字符数 · log n)。

bool alienConsistent(const vector<string>& words, int cnt) {   // 前 cnt 个单词是否相容
    vector<vector<int>> g(26);
    vector<int> indeg(26, 0);
    for (int i = 0; i + 1 < cnt; ++i) {
        const string& a = words[i];
        const string& b = words[i + 1];
        size_t j = 0;
        while (j < a.size() && j < b.size() && a[j] == b[j]) ++j;
        if (j == min(a.size(), b.size())) {
            if (a.size() > b.size()) return false;           // 长的排在前缀前面
            continue;
        }
        g[a[j] - 'a'].push_back(b[j] - 'a');
        ++indeg[b[j] - 'a'];
    }
    queue<int> q;
    for (int c = 0; c < 26; ++c) if (indeg[c] == 0) q.push(c);   // 没出现的字母入度也是 0，不影响判环
    int out = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++out;
        for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
    }
    return out == 26;                                        // 有环时环上的字母出不了队
}
int firstConflict(const vector<string>& words) {
    int n = words.size();
    if (alienConsistent(words, n)) return -1;
    int lo = 0, hi = n - 1;                                  // 找第一个让前 j+1 个矛盾的 j
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (alienConsistent(words, mid + 1)) lo = mid + 1;
        else hi = mid;
    }
    return lo;
}
def alien_consistent(words, cnt):                       # 前 cnt 个单词是否相容

    indeg = [0] * 26

        j = 0
        while j < len(a) and j < len(b) and a[j] == b[j]:
            j += 1

            if len(a) > len(b):                         # 长的排在前缀前面

  • 比较了第一处不同之后的字母：例如从 "ab"、"ba" 推出 b<a，这是错的；只有第一处 a<b 有效。

原题回顾

16.16 说过：Kahn 算法某一步入度为 0 的点不止一个，拓扑序就不唯一。

变化点

LeetCode 444：给出 1..n 的一个排列 nums，以及若干个子序列 seqs（每个都是 nums 的子序列，但不一定连续）。问 nums 是不是唯一一个以所有 seqs 为子序列的最短序列。换成拓扑的话说：seqs 中相邻两个数给出一条先后约束，问在这些约束下拓扑序是否唯一，并且就是 nums。[4]

为什么这样改：在位置 ④ 加两道检查

拓扑序唯一，等价于 Kahn 的每一步队列里恰好有 1 个点：一旦有 2 个点同时可选，它们谁先谁后都满足约束，至少有两种拓扑序。于是在出队前检查 q.size() > 1 就返回 false；出队的点还要和 nums 当前位置对得上。最后检查是否走完了全部 n 个点（有环会提前停下）。

代码

bool uniqueReconstruct(const vector<int>& nums, const vector<vector<int>>& seqs) {   // nums 是 1..n 的排列
    int n = nums.size();
    vector<vector<int>> g(n + 1);
    vector<int> indeg(n + 1, 0);
    for (const auto& s : seqs)
        for (size_t i = 0; i + 1 < s.size(); ++i) {                    // 相邻两个数就是一条边
            g[s[i]].push_back(s[i + 1]);
            ++indeg[s[i + 1]];
        }
    queue<int> q;
    for (int v = 1; v <= n; ++v) if (indeg[v] == 0) q.push(v);
    int idx = 0;
    while (!q.empty()) {
        if (q.size() > 1) return false;                                // 同时有两个可选：不唯一
        int u = q.front(); q.pop();
        if (nums[idx++] != u) return false;                            // 唯一的顺序和 nums 对不上
        for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
    }
    return idx == n;                                                   // 有环会提前停下
}
def unique_reconstruct(nums, seqs):                     # nums 是 1..n 的排列
    n = len(nums)
    g = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    for s in seqs:
        for x, y in zip(s, s[1:]):                      # 相邻两个数就是一条边
            g[x].append(y)
            indeg[y] += 1
    q = deque(v for v in range(1, n + 1) if indeg[v] == 0)
    idx = 0
    while q:
        if len(q) > 1:                                  # 同时有两个可选：不唯一
            return False
        u = q.popleft()
        if nums[idx] != u:                              # 唯一的顺序和 nums 对不上
            return False
        idx += 1
        for v in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return idx == n                                     # 有环会提前停下

参考答案
bool isValidTopo(int n, const vector<pair<int, int>>& edges, const vector<int>& order) {
    if ((int)order.size() != n) return false;
    vector<int> pos(n, -1);
    for (int i = 0; i < n; ++i) {
        int x = order[i];
        if (x < 0 || x >= n || pos[x] != -1) return false;            // 越界或重复
        pos[x] = i;                                                    // pos[x]：x 排在第几
    }
    for (auto [u, v] : edges) if (pos[u] > pos[v]) return false;       // 有一条边方向反了
    return true;
}
def is_valid_topo(n, edges, order):
    if len(order) != n:
        return False
    pos = [-1] * n
    for i, x in enumerate(order):
        if x < 0 or x >= n or pos[x] != -1:             # 越界或重复
            return False
        pos[x] = i                                      # pos[x]：x 排在第几
    return all(pos[u] < pos[v] for u, v in edges)       # 每条边都是前指后

练习 12.14-1【基础】比赛名次能否唯一确定

题面：n 名选手（编号 0..n−1）进行了若干场比赛，results 中的 (a, b) 表示 a 的名次在 b 之前。写 vector<int> uniqueRanking(int n, results)：如果由这些结果能唯一确定全部名次，返回从第 1 名到第 n 名的选手编号；不能唯一确定或结果互相矛盾时返回空数组。

vector<int> uniqueRanking(int n, const vector<pair<int, int>>& results) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [a, b] : results) { g[a].push_back(b); ++indeg[b]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    vector<int> rank;
    while (!q.empty()) {
        if (q.size() > 1) return {};                         // 两人都可以排在这一名：不唯一
        int u = q.front(); q.pop();
        rank.push_back(u);
        for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
    }
    return (int)rank.size() == n ? rank : vector<int>{};     // 不足 n 个：结果成环
}

练习 12.14-2【变形】看相邻两点之间有没有边

题面：另一种判断拓扑序是否唯一的方法：先随便求出一个拓扑序，再检查其中每一对相邻的点 order[i]、order[i+1] 之间是否有边 order[i]→order[i+1]。写 bool uniqueByEdges(int n, edges) 用这种方法判断；有环返回 false。

样例：n = 3，edges = [(0,1),(1,2),(0,2)]，返回 true。n = 3，edges = [(0,1),(0,2)]，返回 false：求得的拓扑序 0 1 2 中，1 和 2 之间没有边。n = 1 没有边，返回 true。

提示

若某一对相邻的 a、b 之间没有边 a→b，交换它们得到的仍是拓扑序：b 的前驱都排在 b 前面，a 又不是 b 的前驱，所以它们都在 a 前面；a 的后继都在 a 后面，b 又不是 a 的后继，所以它们都在 b 后面。于是至少有两个拓扑序。反过来，相邻两点都有边时，这些边连成一条经过所有点的链，链上的先后被边逐个固定，只有一种排法。实现时把边放进 set<pair<int, int>>，每对相邻点查一次。

参考答案
bool uniqueByEdges(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    set<pair<int, int>> has(edges.begin(), edges.end());    // 用来查相邻两点之间有没有边
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    vector<int> order;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
    }
    if ((int)order.size() < n) return false;                 // 有环
    for (int i = 0; i + 1 < n; ++i)
        if (!has.count({order[i], order[i + 1]})) return false;   // 这一对可以交换
    return true;
}
def unique_by_edges(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    has = set(edges)                                    # 用来查相邻两点之间有没有边
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    if len(order) < n:                                  # 有环
        return False
    return all((order[i], order[i + 1]) in has for i in range(n - 1))   # 有一对没边就能交换

练习 12.14-3【综合】哪些点的名次已经固定

题面：名次不唯一时，仍可能有一部分点在所有拓扑序里都排在同一个位置。给 n 个点（n ≤ 100）和边表 edges，写 vector<int> fixedNodes(int n, edges)，按编号升序返回这些点；有环返回空数组。

样例：12.10 的例图 E，输出 [2]：2 的祖先是另外 5 个点，它永远排最后；其余点都至少和一个点没有先后关系。n = 3，edges = [(0,1),(1,2)]，输出 [0,1,2]。n = 4，edges = [(0,1),(1,2)]，输出 []：3 可以插在任何位置，其他三个点的名次随之变动。

提示

由练习 12.10-5，v 最早排在第 anc(v) + 1 位，最晚排在第 n − desc(v) 位，anc 是祖先数，desc 是后代数。两者相等即名次固定，也就是 anc(v) + desc(v) = n − 1：其余每个点都是 v 的祖先或后代。祖先集合用 12.18 的方法沿拓扑序传递：pre[v] |= pre[u]，再把 u 自己加进去。后代数不必再算一遍：u 是 v 的后代，当且仅当 v 在 pre[u] 里，所以数一数有多少个 u 的 pre[u] 含 v。总共 O(n²) 次位查询。

vector<int> fixedNodes(int n, const vector<pair<int, int>>& edges) {   // n ≤ 100
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    vector<bitset<100>> pre(n);                              // pre[v]：v 的全部祖先
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int done = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++done;
        for (int v : g[u]) {
            pre[v] |= pre[u];
            pre[v][u] = 1;
            if (--indeg[v] == 0) q.push(v);
        }
    }
    if (done < n) return {};
    vector<int> ans;
    for (int v = 0; v < n; ++v) {
        int anc = pre[v].count(), desc = 0;
        for (int u = 0; u < n; ++u) if (pre[u][v]) ++desc;   // v 是 u 的祖先，u 就是 v 的后代
        if (anc + desc == n - 1) ans.push_back(v);           // 其余每个点都和 v 有先后
    }
    return ans;
}
def fixed_nodes(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    pre = [0] * n                                       # pre[v]：v 的全部祖先（整数当位集合）
    q = deque(i for i in range(n) if indeg[i] == 0)
    done = 0
    while q:
        u = q.popleft()
        done += 1
        for v in g[u]:
            pre[v] |= pre[u] | (1 << u)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    if done < n:
        return []
    ans = []
    for v in range(n):
        anc = bin(pre[v]).count("1")
        desc = sum(1 for u in range(n) if pre[u] >> v & 1)   # v 是 u 的祖先，u 就是 v 的后代
        if anc + desc == n - 1:                         # 其余每个点都和 v 有先后
            ans.append(v)
    return ans

易错点：

  • 只检查了队列大小，没核对出队的点是否等于 nums[idx]：seqs 只有 [3,1,2,4,5] 时唯一的序列是 3 1 2 4 5，nums = [3,1,4,2,5] 也会被判成 true。
  • 忘了最后的 idx == n：有环时循环会提前结束，前面每一步都没出错。
  • 验证拓扑序时没查重复和越界：[3,1,4,5,0,0] 长度对了，却漏掉了 2。

小结： 拓扑序唯一，当且仅当每一步队列中恰好有 1 个点；序列重建问题还要核对每次出队的点是否等于 nums[idx]；验证一个序列是否为拓扑序时，用名次数组逐条检查边。
