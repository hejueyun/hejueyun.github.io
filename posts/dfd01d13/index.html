11.6 网格动态规划：不同路径与最小路径和 选读

机器人在网格中只能向右或向下走，问路径的条数或最小代价。由于移动方向是单调的，路径不会绕圈，每个格子只能从上方或左方到达，所以一个格子的答案由这两个邻居的答案决定，dp 表的大小与网格相同。

下面的写法中，m、n 是行数和列数，dp[i][j] 是到达格子 (i,j) 的答案。第 0 行和第 0 列的格子只能从一个方向到达，需要单独初始化。

初始化 dp[0][0]、第 0 行、第 0 列
for i in 1..m-1:            # 从上到下
    for j in 1..n-1:        # 从左到右
        dp[i][j] = 合并(dp[i-1][j], dp[i][j-1]) + 本格代价
answer = dp[m-1][n-1]

62 不同路径

输入网格的行数 m 和列数 n，求从左上角走到右下角、每步只向右或向下的路径条数。设 dp[i][j] 为从 (0,0) 走到 (i,j) 的路径数。到达 (i,j) 的最后一步要么来自上方，要么来自左方，两类路径互不重叠，所以 dp[i][j] = dp[i−1][j] + dp[i][j−1]。按行从上到下、每行从左到右填表，计算 dp[i][j] 时上方和左方的格子都已填好。答案是 dp[m−1][n−1]。

uniquePaths(3, 4)：按行从上到下、每行从左到右

1 / 4

第1步 · 初始化：第 0 行全是 1（只能一路向右），第 0 列也全是 1

2 / 4

第2步 · dp[1][1]=上 1 + 左 1 = 2

3 / 4

第3步 · 第 1 行填完：2、3、4

4 / 4

第4步 · 第 2 行：3、6、10；右下角 dp[2][3]=10 就是答案

int uniquePaths(int m, int n) {
    vector<vector<int>> dp(m, vector<int>(n, 1));      // 第 0 行、第 0 列都是 1
    for (int i = 1; i < m; ++i)
        for (int j = 1; j < n; ++j)
            dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
    return dp[m - 1][n - 1];
}
def unique_paths(m, n):
    dp = [[1] * n for _ in range(m)]                          # 第 0 行、第 0 列都是 1；不能写 [[1] * n] * m：m 行是同一个列表，改 dp[i][j] 时每一行的第 j 格都跟着变
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = dp[i - 1][j] + dp[i][j - 1]
    return dp[m - 1][n - 1]

时间 O(mn)，空间 O(mn)，只保留一行可降到 O(n)。这个问题也有组合数学的解法：从左上到右下一共走 m+n−2 步，其中 m−1 步向下，答案是 C(m+n−2, m−1)。例如 3×4 的网格是 C(5,2) = 10，3×7 的网格是 C(8,2) = 28。

64 最小路径和

输入非负整数网格 grid，求从左上角走到右下角（只向右、向下）的路径上数字之和的最小值。设 dp[i][j] 为从 (0,0) 走到 (i,j) 的最小路径和，包含起点和终点格子。与 62 题相同，最后一步来自上方或左方，取两者中较小的一个再加上当前格子：dp[i][j] = min(dp[i−1][j], dp[i][j−1]) + grid[i][j]。答案是 dp[m−1][n−1]。

原网格 grid；最优路径 1→3→1→1→1，和为 7

minPathSum：dp 表的填写

1 / 5

第1步 · dp[0][0]=1；第 0 行只能从左来：1,4,5；第 0 列只能从上来：1,2,6

2 / 5

第2步 · dp[1][1]=min(上 4, 左 2)+5=7

3 / 5

第3步 · dp[1][2]=min(上 5, 左 7)+1=6

4 / 5

第4步 · dp[2][1]=min(上 7, 左 6)+2=8

5 / 5

第5步 · dp[2][2]=min(上 6, 左 8)+1=7；沿着“取了哪个 min”倒推就是路径

int minPathSum(const vector<vector<int>>& grid) {
    int m = grid.size(), n = grid[0].size();
    vector<vector<int>> dp(m, vector<int>(n));
    dp[0][0] = grid[0][0];
    for (int j = 1; j < n; ++j) dp[0][j] = dp[0][j - 1] + grid[0][j];
    for (int i = 1; i < m; ++i) dp[i][0] = dp[i - 1][0] + grid[i][0];
    for (int i = 1; i < m; ++i)
        for (int j = 1; j < n; ++j)
            dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
    return dp[m - 1][n - 1];
}
def min_path_sum(grid):
    m, n = len(grid), len(grid[0])
    dp = [[0] * n for _ in range(m)]                          # 列表推导式：每行都是独立的新列表
    dp[0][0] = grid[0][0]
    for j in range(1, n): dp[0][j] = dp[0][j - 1] + grid[0][j]
    for i in range(1, m): dp[i][0] = dp[i - 1][0] + grid[i][0]
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j]
    return dp[m - 1][n - 1]

时间 O(mn)，空间 O(mn)。

小结： 网格动态规划的状态就是走到 (i,j) 时的答案。填表的顺序必须保证所依赖的格子已经填好，只能向右、向下走时按行填即可。计数问题用加法合并，最值问题用 min 或 max 合并。

练习 11.6-1【基础】仓库机器人绕开货架

题面：m×n 的仓库网格，0 是通道、1 是货架。机器人从左上角出发，只能向右或向下，到右下角有多少条路径？起点或终点是货架时答案为 0。1 ≤ m, n ≤ 16。样例：[[0,0,0],[0,1,0],[0,0,0]] → 2；[[0,1],[0,0]] → 1；[[1,0]] → 0。

提示

状态：dp[i][j] 表示从起点走到 (i,j) 的路径数。转移：来自上方和左方的路径数相加；货架格子走不进去，路径数为 0。和 62 相比改一处：首行首列遇到货架后，后面的格子也到不了，所以不能一次性全设成 1。更稳的写法是统一在双重循环里处理："是货架就置 0；是起点就置 1；否则有上加上、有左加左"。相近题型：LeetCode 63。

int robotPaths_6_1(const vector<vector<int>>& g) {
    int m = g.size(), n = g[0].size();
    vector<vector<int>> dp(m, vector<int>(n, 0));
    for (int i = 0; i < m; ++i)
        for (int j = 0; j < n; ++j) {
            if (g[i][j] == 1) continue;                // 货架：保持 0
            if (i == 0 && j == 0) { dp[i][j] = 1; continue; }
            if (i > 0) dp[i][j] += dp[i - 1][j];
            if (j > 0) dp[i][j] += dp[i][j - 1];
        }
    return dp[m - 1][n - 1];
}
def robot_paths_6_1(g):
    m, n = len(g), len(g[0])
    dp = [[0] * n for _ in range(m)]
    for i in range(m):
        for j in range(n):
            if g[i][j] == 1: continue                         # 货架：保持 0
            if i == 0 and j == 0:
                dp[i][j] = 1
                continue
            if i > 0: dp[i][j] += dp[i - 1][j]
            if j > 0: dp[i][j] += dp[i][j - 1]
    return dp[m - 1][n - 1]

练习 11.6-2【变形】数字三角形自底向上

题面：数字三角形第 i 行（从 0 开始）有 i+1 个数。从顶部出发，每步走到下一行相邻的两个位置之一（下标 j 或 j+1），求走到最底行的最小路径和。1 ≤ 行数 ≤ 200，−10^4 ≤ 值 ≤ 10^4。样例：[[2],[3,4],[6,5,7],[4,1,8,3]] → 11（2+3+5+1）；[[-10]] → −10。

提示

状态：dp[j] 表示从当前行第 j 个数出发、走到最底行的最小路径和。这样定义是自底向上：答案落在唯一的顶点上，省掉了"底行哪个终点最好"的比较。转移：从下一行的两个邻居 j、j+1 中挑更小的那条路，再加上自己。和 64 相比：网格变成三角形，填表方向从自上而下改成自下而上，而且一维数组就够了。相近题型：LeetCode 120。

int triangleMin_6_2(const vector<vector<int>>& t) {
    int n = t.size();
    vector<int> dp(t[n - 1]);                          // 最底行：从自己出发，和就是自己
    for (int i = n - 2; i >= 0; --i)
        for (int j = 0; j <= i; ++j)
            dp[j] = min(dp[j], dp[j + 1]) + t[i][j];   // dp[j]、dp[j+1] 此时还是下一行的值
    return dp[0];
}
def triangle_min_6_2(t):
    n = len(t)
    dp = t[n - 1][:]                                          # 最底行：从自己出发，和就是自己；切片复制，别改动输入
    for i in range(n - 2, -1, -1):
        for j in range(i + 1):
            dp[j] = min(dp[j], dp[j + 1]) + t[i][j]           # dp[j]、dp[j+1] 此时还是下一行的值
    return dp[0]

练习 11.6-3【综合】地图上最大的正方形空地

题面：m×n 地图由 '0'（有建筑）和 '1'（空地）组成。找出只含 '1' 的最大正方形，返回它的面积。1 ≤ m, n ≤ 300。样例：["10100","10111","11111","10010"] → 4；["0"] → 0；["11","11"] → 4。

提示

状态：dp[i][j] 表示以 (i,j) 为右下角、全是 '1' 的最大正方形的边长。转移：本格是 '0' 时为 0；本格是 '1' 时，边长受上方、左方、左上方三个邻居共同限制——想一想三者里谁是瓶颈，再加上本格这一圈。易错：答案要的是面积（边长的平方），不是边长；dp 多开一行一列可以免去首行首列的特判。对拍：枚举左上角和边长，逐格检查是否全是 '1'。相近题型：LeetCode 221。

参考答案
int maxSquare_6_3(const vector<string>& g) {
    int m = g.size(), n = g[0].size(), side = 0;
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));   // dp[i][j] 对应格子 (i-1, j-1)
    for (int i = 1; i <= m; ++i)
        for (int j = 1; j <= n; ++j)
            if (g[i - 1][j - 1] == '1') {
                dp[i][j] = 1 + min({dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]});
                side = max(side, dp[i][j]);
            }
    return side * side;
}
def max_square_6_3(g):
    m, n, side = len(g), len(g[0]), 0
    dp = [[0] * (n + 1) for _ in range(m + 1)]                # dp[i][j] 对应格子 (i-1, j-1)
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if g[i - 1][j - 1] == '1':
                dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
                side = max(side, dp[i][j])
    return side * side

易错点：

  • 第 0 行、第 0 列忘记单独初始化，访问 dp[−1][j] 越界。
  • 64 题把第 0 行全设为 grid 的值而不累加：第 0 行的 dp 是前缀和。

11.7 两个字符串的动态规划：最长公共子序列与编辑距离 选读

比较两个字符串的问题（如文本差异比较、拼写纠错、DNA 序列比对），输入是两个字符串或两个数组，要求公共部分的长度、转换的代价或匹配的方式。这时状态要同时描述第一个串处理到哪里、第二个串处理到哪里，所以有两个下标 i、j，dp 表的大小是 (m+1)×(n+1)。

dp[i][j] 表示 a 的前 i 个字符与 b 的前 j 个字符之间的答案。多出来的第 0 行和第 0 列对应空串，用来存放初始值。转移只需考察两个前缀的最后一个字符 a[i-1] 和 b[j-1]，分它们相等和不相等两种情况讨论。

1143 最长公共子序列（LCS）

从一个序列中删去若干个字符、保持其余字符的相对顺序，得到的序列叫子序列，它不要求连续。例如 "ace" 是 "abcde" 的子序列。本题求两个字符串的最长公共子序列的长度。

设 dp[i][j] 为 a 的前 i 个字符与 b 的前 j 个字符的最长公共子序列长度，空串与任何串的公共子序列长度为 0，所以 dp[0][] 和 dp[][0] 都是 0。如果 a[i−1] == b[j−1]，这个字符可以作为公共子序列的最后一个字符，dp[i][j] = dp[i−1][j−1] + 1；否则两个末尾字符中至少有一个不在最长公共子序列中，dp[i][j] = max(dp[i−1][j], dp[i][j−1])。按 i、j 递增的顺序填表，计算 dp[i][j] 时它左方、上方、左上方的值都已算好。答案是 dp[m][n]。

int longestCommonSubsequence(const string& a, const string& b) {
    int m = a.size(), n = b.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (int i = 1; i <= m; ++i)
        for (int j = 1; j <= n; ++j)
            dp[i][j] = (a[i - 1] == b[j - 1]) ? dp[i - 1][j - 1] + 1 : max(dp[i - 1][j], dp[i][j - 1]);
    return dp[m][n];
}
string lcsString(const string& a, const string& b) {
    int m = a.size(), n = b.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (int i = 1; i <= m; ++i)
        for (int j = 1; j <= n; ++j)
            dp[i][j] = (a[i - 1] == b[j - 1]) ? dp[i - 1][j - 1] + 1 : max(dp[i - 1][j], dp[i][j - 1]);
    string out;
    int i = m, j = n;
    while (i > 0 && j > 0) {                           // 从右下角倒着走回去
        if (a[i - 1] == b[j - 1]) { out.push_back(a[i - 1]); --i; --j; }
        else if (dp[i - 1][j] >= dp[i][j - 1]) --i;
        else --j;
    }
    reverse(out.begin(), out.end());
    return out;
}

def lcs_string(a, b):

    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):

    out = []                                                  # 先收集字符到列表，最后 join
    i, j = m, n
    while i > 0 and j > 0:                                    # 从右下角倒着走回去
        if a[i - 1] == b[j - 1]: out.append(a[i - 1]); i -= 1; j -= 1
        elif dp[i - 1][j] >= dp[i][j - 1]: i -= 1
        else: j -= 1
    out.reverse()
    return "".join(out)

求把字符串 a 变成 b 所需的最少操作数，可用的操作是插入、删除、替换一个字符。设 dp[i][j] 为把 a 的前 i 个字符变成 b 的前 j 个字符所需的最少操作数。初始值 dp[i][0] = i（删去全部 i 个字符），dp[0][j] = j（插入 j 个字符）。如果 a[i−1] == b[j−1]，最后一个字符不需要操作，dp[i][j] = dp[i−1][j−1]。否则考虑最后一步做了什么操作：替换 a[i−1] 为 b[j−1]，对应左上方 dp[i−1][j−1]；删除 a[i−1]，对应上方 dp[i−1][j]；在末尾插入 b[j−1]，对应左方 dp[i][j−1]。三者取最小值再加 1。答案是 dp[m][n]。

int minDistance(const string& a, const string& b) {
    int m = a.size(), n = b.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1));
    for (int i = 0; i <= m; ++i) dp[i][0] = i;         // 删光 i 个字符
    for (int j = 0; j <= n; ++j) dp[0][j] = j;         // 插入 j 个字符
    for (int i = 1; i <= m; ++i)
        for (int j = 1; j <= n; ++j) {
            if (a[i - 1] == b[j - 1]) dp[i][j] = dp[i - 1][j - 1];
            else dp[i][j] = 1 + min({dp[i - 1][j - 1],   // 替换
                                     dp[i - 1][j],       // 删除 a[i-1]
                                     dp[i][j - 1]});     // 插入 b[j-1]
        }
    return dp[m][n];
}
def min_distance(a, b):

bool wildMatch_7_5(const string& s, const string& p) {
    int m = s.size(), n = p.size();
    vector<vector<bool>> dp(m + 1, vector<bool>(n + 1, false));
    dp[0][0] = true;
    for (int j = 1; j <= n; ++j) dp[0][j] = dp[0][j - 1] && p[j - 1] == '*';   // 空串只能被一串 '*' 匹配
    for (int i = 1; i <= m; ++i)
        for (int j = 1; j <= n; ++j) {
            if (p[j - 1] == '*') dp[i][j] = dp[i][j - 1] || dp[i - 1][j];       // 匹配空串 / 再吃一个字符
            else if (p[j - 1] == '?' || p[j - 1] == s[i - 1]) dp[i][j] = dp[i - 1][j - 1];
        }
    return dp[m][n];
}

            elif p[j - 1] == '?' or p[j - 1] == s[i - 1]: dp[i][j] = dp[i - 1][j - 1]
    return dp[m][n]

易错点：

  • 下标错位：dp 的 i 对应字符 a[i-1]。
  • 混淆子序列和子串：子串要求连续，"最长公共子串"的转移在不等时置 0，而不是取 max。

11.8 背包问题：0-1 背包、完全背包与四道变形题 必读

从一组物品中挑选一些，使总重量不超过容量并且总价值最大，这是动态规划的经典模型，称为背包问题。很多题目换了说法，例如凑硬币、分割子集、拆分成完全平方数，本质仍是背包问题。17.23 已经用“重 2 价 4、重 3 价 5、重 4 价 7、容量 5”的例子画出过二维表，并给出了贪心地选最贵物品的反例；本节在此基础上讲解一维写法和四道变形题。[1]

一道题可以看作背包问题，是指它有一组物品（数、硬币、平方数）和一个容量或目标（重量上限、金额、和），每个物品要么选或不选（0-1 背包），要么可以选任意次（完全背包），要求最大价值、最少个数、方案数或者能否恰好凑出目标。

0-1 背包：从二维到一维

有 n 件物品，第 i 件的重量为 w、价值为 v，背包容量为 cap，每件物品最多选一次，求最大总价值。设 dp[i][c] 为只考虑前 i 件物品、容量为 c 时的最大价值。第 0 行（没有物品）全为 0。对第 i 件物品：不选，结果是 dp[i−1][c]；选（要求 c ≥ w），结果是 dp[i−1][c−w] + v；两者取较大值。第 i 行只依赖第 i−1 行，答案是 dp[n][cap]。

0-1 背包二维表（与 17.6 同一组数据）：dp[2][5]=max(上一行 dp[1][5]=4, 上一行 dp[1][2]+5=9)=9。每一格只看“上一行”（图外灰色小数字是图的行列号，dp 下标 = 行列号 − 1）

int knapsack01_2d(const vector<int>& w, const vector<int>& v, int cap) {
    int n = w.size();
    vector<vector<int>> dp(n + 1, vector<int>(cap + 1, 0));
    for (int i = 1; i <= n; ++i)
        for (int c = 0; c <= cap; ++c) {
            dp[i][c] = dp[i - 1][c];                                    // 不选第 i 件
            if (c >= w[i - 1]) dp[i][c] = max(dp[i][c], dp[i - 1][c - w[i - 1]] + v[i - 1]);  // 选
        }
    return dp[n][cap];
}
def knapsack01_2d(w, v, cap):
    n = len(w)
    dp = [[0] * (cap + 1) for _ in range(n + 1)]              # 每行独立；[[0] * (cap + 1)] * (n + 1) 会让"上一行"和"这一行"是同一个列表
    for i in range(1, n + 1):
        for c in range(cap + 1):
            dp[i][c] = dp[i - 1][c]                                                         # 不选第 i 件
            if c >= w[i - 1]: dp[i][c] = max(dp[i][c], dp[i - 1][c - w[i - 1]] + v[i - 1])  # 选
    return dp[n][cap]

压缩成一维。 由于每一行只用到上一行，可以只保留一行 dp[c]，在原地把上一行的值改写成这一行的值。困难在于：计算 dp[c] 时要读取 dp[c−w]，它必须仍是上一行的值。

  • 倒序（c 从 cap 递减到 w）：更新 dp[c] 时，下标更小的 dp[c−w] 在本轮还没被修改，仍是上一行的值，所以第 i 件物品最多被选一次。
  • 正序：dp[c−w] 已经在本轮被修改过，其中可能已经放入了第 i 件物品，再放一次就重复使用了这件物品。

    vector<int> dp(cap + 1, 0);
    for (int i = 0; i < (int)w.size(); ++i)

完全背包中每件物品可以选任意次。二维转移变为 dp[i][c] = max(dp[i−1][c], dp[i][c−w] + v)：选了第 i 件之后还可以继续选第 i 件，所以读的是本行的 dp[i][c−w]。一维写法中正序遍历，读到的恰好是本行已经更新过的值。

int completeKnapsack(const vector<int>& w, const vector<int>& v, int cap) {
    vector<int> dp(cap + 1, 0);
    for (int i = 0; i < (int)w.size(); ++i)
        for (int c = w[i]; c <= cap; ++c)              // 正序：允许本轮再次选同一件
            dp[c] = max(dp[c], dp[c - w[i]] + v[i]);
    return dp[cap];
}
def complete_knapsack(w, v, cap):
    dp = [0] * (cap + 1)
    for i in range(len(w)):
        for c in range(w[i], cap + 1):                        # 正序：允许本轮再次选同一件
            dp[c] = max(dp[c], dp[c - w[i]] + v[i])
    return dp[cap]

322 零钱兑换（最少硬币数）

输入硬币面额 coins（每种数量无限）和金额 amount，求凑出 amount 所需的最少硬币数，凑不出时返回 −1。设 dp[x] 为恰好凑出金额 x 所需的最少硬币数，凑不出时为无穷大；由于答案最多是 amount 枚，用 amount+1 表示无穷大即可。dp[0] = 0。按 x 从小到大计算，枚举最后一枚硬币的面额 c：dp[x] = min(dp[x], dp[x−c] + 1)。最后若 dp[amount] 大于 amount，说明凑不出，返回 −1。

2 / 5

第2步 · x=1..4：dp[3]=min(dp[2]+1, dp[1]+1)=2，dp[4]=dp[2]+1=2

3 / 5

第3步 · x=5：最后一枚用 5，dp[0]+1=1

4 / 5

第4步 · x=6..10：dp[10]=dp[5]+1=2

5 / 5

第5步 · x=11：候选 dp[10]+1=3、dp[9]+1=4、dp[6]+1=3，取 3

int coinChange(const vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, amount + 1);            // amount+1 充当"无穷大"
    dp[0] = 0;
    for (int x = 1; x <= amount; ++x)
        for (int c : coins)
            if (c <= x) dp[x] = min(dp[x], dp[x - c] + 1);
    return dp[amount] > amount ? -1 : dp[amount];
}
def coin_change(coins, amount):
    dp = [amount + 1] * (amount + 1)                          # amount+1 充当"无穷大"；Python 也可以用 float('inf')，+1 不会溢出
    dp[0] = 0
    for x in range(1, amount + 1):
        for c in coins:
            if c <= x: dp[x] = min(dp[x], dp[x - c] + 1)
    return -1 if dp[amount] > amount else dp[amount]

时间 O(amount·|coins|)，空间 O(amount)。求最小值时，内外层循环的先后顺序都可以，因为最小个数与硬币的使用顺序无关。

518 零钱兑换 II（组合数）

输入 amount 和 coins（每种数量无限），求凑出 amount 的组合数，1+2 与 2+1 算作同一种。设 dp[x] 为只用已经处理过的硬币凑出 x 的组合数，dp[0] = 1，表示什么都不拿也是一种方法。外层循环逐个加入硬币；加入硬币 c 后，内层正序让 x 从 c 增加到 amount，执行 dp[x] += dp[x−c]（完全背包，c 可以使用多次）。答案是 dp[amount]。

外层循环是物品还是金额，结果不同。 外层是硬币时，硬币按固定的顺序加入，每种组合只会以一种顺序（按硬币加入的先后）被计数一次，得到的是组合数。外层是金额时，每个金额都重新枚举最后一枚是哪种硬币，1+2 和 2+1 的最后一枚不同，会被分别计数，得到的是排列数，那是 377 题“组合总和 IV”的问法。

int change(int amount, const vector<int>& coins) {
    vector<long long> dp(amount + 1, 0);
    dp[0] = 1;
    for (int c : coins)                                // 外层物品：组合数
        for (int x = c; x <= amount; ++x)
            dp[x] += dp[x - c];
    return (int)dp[amount];
}

416 分割等和子集

bool canPartition(const vector<int>& nums) {
    int sum = 0;
    for (int x : nums) sum += x;
    if (sum % 2 != 0) return false;
    int target = sum / 2;
    vector<bool> dp(target + 1, false);
    dp[0] = true;
    for (int x : nums)
        for (int s = target; s >= x; --s)              // 0-1 背包：倒序
            dp[s] = dp[s] || dp[s - x];
    return dp[target];
}
def can_partition(nums):
    total = sum(nums)                                         # 变量不叫 sum，免得遮住内置函数 sum
    if total % 2 != 0: return False
    target = total // 2                                       # // 是整除；/ 会得到 float
    dp = [False] * (target + 1)
    dp[0] = True
    for x in nums:
        for s in range(target, x - 1, -1):                    # 0-1 背包：倒序
            dp[s] = dp[s] or dp[s - x]
    return dp[target]

时间 O(n·target)，空间 O(target)。

279 完全平方数

输入正整数 n，求和为 n 的完全平方数的最少个数。设 dp[i] 为和为 i 的完全平方数的最少个数，dp[0] = 0。i 从 1 递增，枚举最后一个平方数 j²（j² ≤ i）：dp[i] = min(dp[i], dp[i−j²] + 1)。这相当于物品是各个平方数、每个可以重复使用的完全背包，与 322 题的结构相同。答案是 dp[n]。

int numSquares(int n) {
    vector<int> dp(n + 1, INT_MAX);
    dp[0] = 0;
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j * j <= i; ++j)
            dp[i] = min(dp[i], dp[i - j * j] + 1);
    return dp[n];
}
def num_squares(n):
    dp = [float('inf')] * (n + 1)                             # Python 没有 INT_MAX，用 float('inf')；inf + 1 仍是 inf，不会溢出
    dp[0] = 0
    for i in range(1, n + 1):
        j = 1
        while j * j <= i:                                     # C++ 的 for (j = 1; j * j <= i; ++j)
            dp[i] = min(dp[i], dp[i - j * j] + 1)
            j += 1
    return dp[n]

时间 O(n√n)，空间 O(n)。

int boundedKnap_8_3(const vector<int>& w, const vector<int>& v, const vector<int>& cnt, int cap) {
    vector<int> dp(cap + 1, 0);
    for (int i = 0; i < (int)w.size(); ++i)
        for (int c = cap; c >= w[i]; --c)              // 倒序：dp[c-k*w] 还没考虑第 i 种

int packBits_8_4(const vector<string>& strs, int m, int n) {
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (const string& s : strs) {
        int z = count(s.begin(), s.end(), '0'), o = (int)s.size() - z;
        for (int i = m; i >= z; --i)                   // 两维都倒序：每条串最多存一次
            for (int j = n; j >= o; --j)
                dp[i][j] = max(dp[i][j], dp[i - z][j - o] + 1);
    }
    return dp[m][n];
}

11.9 子序列的动态规划：最长递增子序列与单词拆分 选读

子序列可以跳过元素选取，所以状态通常定义为以 i 结尾，而转移时要回头考察前面所有可能的位置 j，朴素写法的时间是 O(n²)。题目中出现子序列（可以不连续，但保持原顺序）、前 i 个字符能否被切分等说法，或者当前答案依赖前面任意位置而不只是 i−1 时，就属于这一类。300 题还有一种 O(n log n) 的经典写法，把动态规划和二分查找结合起来。

300 最长递增子序列（LIS）：O(n²) 写法

输入整数数组 nums，求最长严格递增子序列的长度。设 dp[i] 为以 nums[i] 结尾的最长递增子序列的长度，初值 dp[i] = 1，即只含自己。对每个 j < i，如果 nums[j] < nums[i]，nums[i] 就可以接在以 nums[j] 结尾的递增子序列之后，所以 dp[i] = max(dp[i], dp[j] + 1)。最长递增子序列可能在任何位置结尾，答案是所有 dp[i] 中的最大值。

O(n²) 写法：第 0 行是 nums，第 1 行是 dp[i]（以 nums[i] 结尾的 LIS 长度）。答案是整行的最大值 4，在 dp[6]、dp[7] 两处

int lengthOfLISQuadratic(const vector<int>& nums) {
    int n = nums.size(), best = 0;
    vector<int> dp(n, 1);                              // dp[i]：以 nums[i] 结尾的 LIS 长度
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < i; ++j)
            if (nums[j] < nums[i]) dp[i] = max(dp[i], dp[j] + 1);
        best = max(best, dp[i]);
    }
    return best;
}
def length_of_lis_quadratic(nums):
    n, best = len(nums), 0
    dp = [1] * n                                              # dp[i]：以 nums[i] 结尾的 LIS 长度
    for i in range(n):
        for j in range(i):
            if nums[j] < nums[i]: dp[i] = max(dp[i], dp[j] + 1)
        best = max(best, dp[i])
    return best

时间 O(n²)，空间 O(n)。

300 的贪心加二分写法：O(n log n)

对于长度相同的递增子序列，结尾越小越好，因为后面的数更容易接在它之后。据此维护数组 tails：tails[k] 是目前扫描过的元素中，长度为 k+1 的递增子序列的最小结尾。可以证明 tails 始终严格递增，因此可以在它上面二分查找。

每次处理新元素 x 之前，tails 严格递增，且它的长度等于目前的最长递增子序列长度。用 lower_bound 找到第一个 ≥ x 的位置 k：如果 k 等于 tails 的长度，说明 x 比所有结尾都大，可以接在最长的子序列之后，把 x 追加到末尾，长度增加 1；否则把 tails[k] 替换为更小的 x，长度不变，但长度为 k+1 的子序列的结尾变小了。扫描完毕后返回 tails 的长度。注意 tails 本身不一定是一个真实的递增子序列，它只记录各个长度的最小结尾。

int lengthOfLIS(const vector<int>& nums) {
    vector<int> tails;                                 // tails[k]：长度 k+1 的递增子序列的最小结尾
    for (int x : nums) {
        auto it = lower_bound(tails.begin(), tails.end(), x);
        if (it == tails.end()) tails.push_back(x);     // 比所有结尾都大：变长
        else *it = x;                                  // 替换第一个 >= x 的结尾：变小
    }
    return tails.size();
}
from bisect import bisect_left

def length_of_lis(nums):
    tails = []                                    # tails[k]：长度 k+1 的递增子序列的最小结尾
    for x in nums:
        k = bisect_left(tails, x)                 # bisect_left 就是 lower_bound：第一个 >= x 的位置（返回下标，不是迭代器）
        if k == len(tails):
            tails.append(x)                       # 比所有结尾都大：变长
        else:
            tails[k] = x                          # 替换第一个 >= x 的结尾：变小
    return len(tails)

每个元素做一次二分查找，时间 O(n log n)，空间 O(n)。

bool wordBreak(const string& s, const vector<string>& wordDict) {
    unordered_set<string> dict(wordDict.begin(), wordDict.end());
    int n = s.size();
    vector<bool> dp(n + 1, false);                     // dp[i]：前 i 个字符能否拆分
    dp[0] = true;
    for (int i = 1; i <= n; ++i)
        for (int j = 0; j < i && !dp[i]; ++j)
            if (dp[j] && dict.count(s.substr(j, i - j))) dp[i] = true;
    return dp[n];
}
def word_break(s, word_dict):
    words = set(word_dict)                                    # unordered_set → set；不叫 dict，免得遮住内置类型
    n = len(s)
    dp = [False] * (n + 1)                                    # dp[i]：前 i 个字符能否拆分
    dp[0] = True
    for i in range(1, n + 1):
        for j in range(i):
            if dp[i]: break                                   # 对应 C++ 循环条件里的 !dp[i]
            if dp[j] and s[j:i] in words: dp[i] = True        # 切片 s[j:i] 写的是终点（不含）；substr(j, i-j) 写的是长度
    return dp[n]

两层循环共有 O(n²) 对 (j, i)，每次截取子串并计算哈希需要 O(n)，总时间 O(n³)。如果限制单词的最大长度为 L，只需枚举 i−L ≤ j < i，时间可降到 O(n·L²)。

小结： 最长递增子序列的 O(n²) 写法中状态是以 i 结尾，答案取所有状态的最大值；O(n log n) 写法维护每个长度的最小结尾，用 lower_bound 找到替换位置。求严格递增时用 lower_bound（第一个 ≥ x 的位置），求非严格递增（允许相等）时改用 upper_bound（第一个 > x 的位置）。139 题的状态是前 i 个字符能否切分，转移时枚举最后一段的起点。

练习 11.9-3【综合】套娃信封

题面：有 n 个信封，用 (宽, 高) 表示。一个信封能装进另一个，当且仅当宽和高都严格更小（不能旋转）。最多能套几层？1 ≤ n ≤ 10^5。样例：[(5,4),(6,4),(6,7),(2,3)] → 3（(2,3)→(5,4)→(6,7)）；[(1,1),(1,1),(1,1)] → 1；[(4,5),(4,6),(6,7),(2,3),(1,1)] → 4。

提示

二维变一维：按宽升序排好后，只需在高度序列上找最长严格递增子序列。易错点在宽相同的信封：(6,4) 和 (6,7) 不能互套，但如果同宽按高升序排，4 < 7 会被当成能接上。解决办法：同宽时按高降序排，这样同宽信封在高度序列里是递减的，最多只会被选进一个。之后用本节的 tails + lower_bound（严格递增用 lower_bound），O(n log n)。对拍：n ≤ 10 时枚举所有信封子集，按宽排序后检查能否一层层严格套住。相近题型：LeetCode 354。

参考答案
int nestBoxes_9_3(vector<pair<int, int>> e) {
    sort(e.begin(), e.end(), [](const pair<int, int>& x, const pair<int, int>& y) {
        return x.first != y.first ? x.first < y.first : x.second > y.second;   // 宽升序，同宽高降序
    });
    vector<int> tails;                                 // tails[k]：长度 k+1 的递增高度序列的最小结尾
    for (const auto& p : e) {
        auto it = lower_bound(tails.begin(), tails.end(), p.second);
        if (it == tails.end()) tails.push_back(p.second);
        else *it = p.second;
    }
    return (int)tails.size();
}
def nest_boxes_9_3(e):
    e = sorted(e, key=lambda p: (p[0], -p[1]))                # 宽升序，同宽高降序
    tails = []                                                # tails[k]：长度 k+1 的递增高度序列的最小结尾
    for w, h in e:
        k = bisect_left(tails, h)                             # 等价于 lower_bound；需要 from bisect import bisect_left
        if k == len(tails): tails.append(h)
        else: tails[k] = h
    return len(tails)
