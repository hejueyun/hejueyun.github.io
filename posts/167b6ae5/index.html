11.1 回溯的一般写法：路径、选择列表、结束条件

当题目要求返回所有组合、列出全部排列或者给出所有合法的棋盘摆法时，答案本身的个数就可能是指数级的，只能逐个构造出来。回溯就是系统地尝试每一种选择：做出一个选择后继续向下尝试；一条路试完后退回来，换下一个选择。

下列情况通常提示应当使用回溯：

  • 输出是所有方案（如 vector<vector<int>>、vector<string>），而不是一个数；
  • 数据规模很小，n 在 20 左右以内（见 1.13 的复杂度表：O(2ⁿ)、O(n!) 的算法只在这个规模下可行）；
  • 题目描述的是逐步决策的过程，例如放置、切分、填满、组合字母。

可以把整个搜索过程画成一棵树，每个节点代表一个已经做出的选择序列。写回溯时要确定三个要素：

要素含义在代码里
路径 path从根走到当前节点所做的选择vector<int>& path
选择列表当前节点还能往下走的分支for 循环的范围（start 起、未用过的、未被攻击的…）
结束条件走到叶子，path 已是一个完整答案函数开头的 if (...) { ans.push_back(path); return; }

以 77 组合（从 1..n 选 k 个数）为例，n=4、k=2 的递归树如下：

combine(4,2) 的递归树：每个节点是一条路径 path，边是一次选择；绿色叶子满足结束条件（长度 2）；[4] 后面只剩 0 个数，凑不满 2 个，被循环上界剪掉

调用栈怎样逐帧进入和返回、为什么要用 pop_back 撤销、排列为什么用 used 而组合用 start，8.5 和 8.6 已经逐帧讲过，这里直接使用它们的结论：多个调用共享的 path 必须在子调用返回后撤销；组合类问题用 start 使选出的下标递增，从而避免同一组数以不同顺序重复出现。

下面是组合问题的通用写法。n、k 是题目参数；start 是本层第一个可以选的数；path 是当前路径；ans 收集完整答案的副本；i 是本层正在尝试的选择，它是本层调用的局部变量。

void dfs77(int n, int k, int start, vector<int>& path, vector<vector<int>>& ans) {
    if ((int)path.size() == k) { ans.push_back(path); return; }          // 结束条件
    for (int i = start; i <= n - (k - (int)path.size()) + 1; ++i) {      // 选择列表（含剪枝）
        path.push_back(i);                                              // 做选择
        dfs77(n, k, i + 1, path, ans);                                  // 进入下一层
        path.pop_back();                                                // 撤销选择
    }
}
vector<vector<int>> combine(int n, int k) {
    vector<int> path; vector<vector<int>> ans;
    dfs77(n, k, 1, path, ans);
    return ans;
}
def dfs77(n, k, start, path, ans):
    if len(path) == k:                                        # 结束条件
        ans.append(path[:])                                   # path[:] 存副本；直接 append(path) 存的是同一个列表，最后全被撤销成空
        return
    for i in range(start, n - (k - len(path)) + 2):           # 选择列表（含剪枝）：i <= n-(k-size)+1，range 右端不含所以写 +2
        path.append(i)                                        # 做选择
        dfs77(n, k, i + 1, path, ans)                         # 进入下一层
        path.pop()                                            # 撤销选择

def combine(n, k):
    path, ans = [], []
    dfs77(n, k, 1, path, ans)
    return ans

剪枝是指提前判断出某个分支中不可能有答案，从而不进入这个分支。最常用的剪枝有两种：

  1. 数量剪枝：path 中已有 size 个数，还需要 k-size 个，那么本层的 i 最大只能取到 n-(k-size)+1，再大的话后面剩下的数不够用。图中通向 [4] 的虚线分支就是这样被剪掉的。
  2. 排序后按值剪枝：先把候选排序，如果 nums[i] 已经超过剩余的目标值，后面更大的数也一定超过，直接 break（11.2 的 40 题会用到）。

剪枝只减少尝试的次数，不改变答案的集合。检验一个剪枝是否正确，只需确认：被剪掉的子树中确实没有答案。

combine(n,k) 的叶子数是 C(n,k)，每个答案复制需要 O(k)，总时间 O(k·C(n,k))；递归深度为 k，不计答案时额外空间为 O(k)。

小结： 回溯就是在决策树上做深度优先搜索。做选择、递归、撤销选择三行代码必须成对出现；剪枝写在循环的范围或循环开头的判断中，作用是减少进入的分支，而不改变答案。

单选 · 理解检查 1　combine(5, 3) 返回多少个组合？在 path 为空时，带数量剪枝的循环中 i 最大取到几？

A. 10 个；i 最大到 3

B. 10 个；i 最大到 5

C. 60 个；i 最大到 3

D. 20 个；i 最大到 4

查看答案与每个选项的解析

A · 正确：C(5,3)=10；还差 3 个数，i ≤ 5−3+1=3，选 4 开头只剩 5 一个数，凑不满。

B · 错误：个数对，但 i=4、5 的分支凑不满 3 个，会被剪掉。

C · 错误：60 是排列数 A(5,3)；组合用 start 保证递增，不区分顺序。

D · 错误：C(5,3)=10；上界公式是 n−(k−size)+1。

void dfsPickCards_1_1(int k, int remain, int start, vector<int>& path, vector<vector<int>>& ans) {
    if ((int)path.size() == k) {                       // 结束条件：张数够了
        if (remain == 0) ans.push_back(path);
        return;
    }

        dfsPickCards_1_1(k, remain - i, i + 1, path, ans);
        path.pop_back();
    }
}

}

void dfsStamps_1_2(const vector<int>& faces, int start, int remain, vector<int>& path, vector<vector<int>>& ans) {
    if (remain == 0) { ans.push_back(path); return; }
    for (int i = start; i < (int)faces.size(); ++i) {
        if (faces[i] > remain) break;                  // 已排序，后面更大
        path.push_back(faces[i]);
        dfsStamps_1_2(faces, i, remain - faces[i], path, ans);   // 传 i：同一面值还能再用
        path.pop_back();
    }
}
vector<vector<int>> stampCombos_1_2(vector<int> faces, int target) {
    sort(faces.begin(), faces.end());
    vector<int> path; vector<vector<int>> ans;
    dfsStamps_1_2(faces, 0, target, path, ans);
    return ans;
}
def dfs_stamps_1_2(faces, start, remain, path, ans):
    if remain == 0:
        ans.append(path[:])
        return
    for i in range(start, len(faces)):
        if faces[i] > remain: break                           # 已排序，后面更大
        path.append(faces[i])
        dfs_stamps_1_2(faces, i, remain - faces[i], path, ans)   # 传 i：同一面值还能再用
        path.pop()

def stamp_combos_1_2(faces, target):
    faces = sorted(faces)                                     # C++ 按值传参得到副本，这里用 sorted 生成新列表
    path, ans = [], []
    dfs_stamps_1_2(faces, 0, target, path, ans)
    return ans

练习 11.1-3【综合】还原丢了点号的 IP 地址

题面：日志系统把 IPv4 地址里的点号全丢了，只剩一串数字 s。在 s 中插入 3 个点切成 4 段，每段是 0～255 的整数且没有前导零（"0" 合法，"01"、"00" 不合法）。按回溯的自然顺序输出所有合法地址：第一段长度从 1 试到 3，其次第二段，依此类推。0 ≤ |s| ≤ 20，s 只含数字。样例："25525511135" → ["255.255.11.135","255.255.111.35"]；"0000" → ["0.0.0.0"]；"101023" → ["1.0.10.23","1.0.102.3","10.1.0.23","10.10.2.3","101.0.2.3"]；"123" → []。

提示

回溯：路径 = 已切出的段 segs；选择列表 = 下一段取 1、2、3 个字符；结束条件 = 已切满 4 段（此时还必须恰好用完 s）。和 17 题一样是"每层选一段"，新增的是合法性判断：前导零、超过 255。再加一个剪枝：剩余字符数要在 [剩余段数, 3×剩余段数] 之间，否则直接返回。对拍时用三重循环枚举三个切点作暴力。相近题型：LeetCode 93。

参考答案
bool validSeg_1_3(const string& s, int l, int len) {
    if (len > 1 && s[l] == '0') return false;          // 前导零
    return stoi(s.substr(l, len)) <= 255;
}
void dfsIp_1_3(const string& s, int pos, vector<string>& segs, vector<string>& ans) {
    int n = s.size(), left = 4 - (int)segs.size();
    if (left == 0) {
        if (pos == n) ans.push_back(segs[0] + "." + segs[1] + "." + segs[2] + "." + segs[3]);
        return;
    }
    if (n - pos < left || n - pos > 3 * left) return;  // 剩余字符太少或太多
    for (int len = 1; len <= 3 && pos + len <= n; ++len) {
        if (!validSeg_1_3(s, pos, len)) continue;
        segs.push_back(s.substr(pos, len));
        dfsIp_1_3(s, pos + len, segs, ans);
        segs.pop_back();
    }
}
vector<string> restoreIp_1_3(const string& s) {
    vector<string> segs, ans;
    dfsIp_1_3(s, 0, segs, ans);
    return ans;
}
def valid_seg_1_3(s, l, length):
    if length > 1 and s[l] == '0': return False              # 前导零
    return int(s[l:l + length]) <= 255

def dfs_ip_1_3(s, pos, segs, ans):
    n, left = len(s), 4 - len(segs)
    if left == 0:
        if pos == n: ans.append(".".join(segs))
        return
    if n - pos < left or n - pos > 3 * left: return           # 剩余字符太少或太多
    for length in range(1, 4):
        if pos + length > n: break
        if not valid_seg_1_3(s, pos, length): continue
        segs.append(s[pos:pos + length])
        dfs_ip_1_3(s, pos + length, segs, ans)
        segs.pop()

def restore_ip_1_3(s):
    segs, ans = [], []
    dfs_ip_1_3(s, 0, segs, ans)
    return ans

易错点：

  • ans.push_back(path) 存的是副本；Python 里要写 ans.append(path[:])，直接 append(path) 存的是同一个列表，最后全被撤销成空。
  • 组合题递归时传 i + 1，写成 start + 1 会让同一层的不同选择都从同一个位置往后选，产生重复和遗漏。

11.2 回溯中的去重：同一层去重，同一条路径保留

候选中有重复数字时，直接使用 11.1 的写法会得到重复的答案。例如从 [1,1,2] 中选出和为 3 的组合，第一个 1 配 2 和第二个 1 配 2 都会得到 [1,2]。去重要达到的目标是：每种答案只产生一次，同时允许一个答案中出现两个相同的数（例如 [1,1,6]）。题目中出现“candidates 中可能有重复元素”“nums 可包含重复数字”“解集不能包含重复的组合”等说法时，就需要去重。

做法是先排序，使相同的数相邻，然后区分两种重复：

  • 同一层的重复：同一个父节点下的两个兄弟分支选了相同的值。它们向下展开的子树完全相同，第二个分支应当剪掉。
  • 同一条路径上的重复：沿着一条路径向下，先选第一个 1，再选第二个 1。这是答案 [1,1,…] 的一部分，应当保留。

排序后 [1,1′,2]、target=3。根这一层第二个 1′ 与第一个 1 是同一层的兄弟，跳过（否则会再得到一次 [1,2]）；[1] 下面选 1′ 是往深处走，属于同一枝，保留

40 组合总和 II

输入是可能含有重复的正整数数组 candidates 和目标 target，输出所有和为 target 的组合，每个下标最多使用一次，组合之间不能重复。nums 是排序后的候选，path 是当前已选的数，remain 是还差多少，start 是本层第一个可选的下标。进入 dfs(start, remain) 时，path 中数的和等于 target - remain，且 path 用过的下标都小于 start。选择 nums[i] 后 remain 减去 nums[i]，下一层从 i+1 开始；返回后弹出 nums[i]，尝试下一个 i。remain 为 0 时记录 path；循环结束说明本层所有选择都已试过，返回。

条件 i > start && nums[i] == nums[i-1] 的含义。 在本层的循环中，i == start 是本层的第一个选择，一定要尝试；i > start 说明 nums[i-1] 也是本层的一个选择，它与 nums[i] 是兄弟。如果两者的值相同，nums[i] 的子树与 nums[i-1] 的子树完全相同，应当跳过。这里必须写 i > start 而不是 i > 0：当 i == start 且 nums[i] == nums[i-1] 时，nums[i-1] 是上一层选择的数，处在同一条路径上，这正是得到 [1,1,6] 所需要的情况。

void dfs40(const vector<int>& nums, int start, int remain, vector<int>& path, vector<vector<int>>& ans) {
    if (remain == 0) { ans.push_back(path); return; }
    for (int i = start; i < (int)nums.size(); ++i) {
        if (nums[i] > remain) break;                          // 剪枝：已排序，后面只会更大
        if (i > start && nums[i] == nums[i - 1]) continue;    // 同层去重
        path.push_back(nums[i]);
        dfs40(nums, i + 1, remain - nums[i], path, ans);      // i+1：每个元素只用一次
        path.pop_back();
    }
}

void dfs47(const vector<int>& nums, vector<bool>& used, vector<int>& path, vector<vector<int>>& ans) {
    if (path.size() == nums.size()) { ans.push_back(path); return; }
    for (int i = 0; i < (int)nums.size(); ++i) {
        if (used[i]) continue;                                            // 这条路径已用过下标 i
        if (i > 0 && nums[i] == nums[i - 1] && !used[i - 1]) continue;    // 同层去重
        used[i] = true; path.push_back(nums[i]);
        dfs47(nums, used, path, ans);
        path.pop_back(); used[i] = false;
    }
}

    return ans;

def dfs47(nums, used, path, ans):
    if len(path) == len(nums):
        ans.append(path[:])
        return
    for i in range(len(nums)):
        if used[i]: continue                                              # 这条路径已用过下标 i
        if i > 0 and nums[i] == nums[i - 1] and not used[i - 1]: continue # 同层去重
        used[i] = True; path.append(nums[i])
        dfs47(nums, used, path, ans)
        path.pop(); used[i] = False

def permute_unique(nums):
    nums.sort()
    used = [False] * len(nums)                                # 一维列表用 * 复制没问题：元素 bool 不可变

    dfs47(nums, used, path, ans)
    return ans

小结： 去重时先排序，使相同的值相邻，然后跳过与前一个值相等、且前一个是兄弟分支的选择。组合类问题用 i > start 判断前一个是否为兄弟，排列类问题用 !used[i-1] 判断。

51 N 皇后

在 n×n 的棋盘上放 n 个皇后，使任意两个皇后不在同一行、同一列或同一条斜线上，要求输出所有摆法。

做法是按行放置：每一层递归负责一行，只在这一行放一个皇后，所以同一行的冲突不可能发生。其余三类冲突各用一个布尔数组记录。同一列上的格子列号相同；同一条主对角线（↘）上的格子 row − col 相同；同一条副对角线（↙）上的格子 row + col 相同。为了让编号从 0 开始，主对角线用 row - c + n - 1 编号：

副对角线（↙）row + col 相同row + c2n − 1

主对角线（左上→右下）编号 row−col+n−1：同一条线上 row−col 相同；+n−1 是为了让下标从 0 开始，共 2n−1 条

副对角线（右上→左下）编号 row+col：同一条线上 row+col 相同，范围 0～2n−2

row 是正在决定的行，board 是当前的棋盘，col[c]、diag[d]、anti[a] 分别表示某一列、某条主对角线、某条副对角线上是否已有皇后。进入 dfs(row) 时，前 row 行各有一个皇后且互不攻击，三个数组准确记录了它们占用的列和斜线。在 (row, c) 放置皇后时，先用三个数组在 O(1) 时间内检查冲突；没有冲突就写入 'Q'，把三个标记设为 true，进入第 row+1 行；返回后把它们全部恢复。row == n 时，n 个皇后都已放好，记录棋盘。

11.4 动态规划的五个步骤：以斐波那契数为例

回溯会把每条路径都走一遍。但很多题目只要求一个数：最大值、最小值、方案数或者是否可行，而且不同的路径会走到同一个子问题。把每个子问题的答案保存下来，使每个子问题只计算一次，这就是动态规划（DP）。

一道题适合用动态规划，通常有以下特征：

  • 要求的是一个数：最大值、最小值、方案数，或者是否可行；
  • 大问题的答案可以由若干个规模更小的同类问题的答案算出，这叫最优子结构；
  • 写成暴力递归后，同样的参数会被反复调用，这叫重叠子问题。

解动态规划题时，按顺序回答以下五个问题：

步骤要回答的问题509 斐波那契数
状态定义dp[i] 表示什么？用一句完整的话说明dp[i] 是第 i 个斐波那契数
转移方程dp[i] 怎样由更小的状态算出？dp[i] = dp[i−1] + dp[i−2]
初始值哪些状态不能由转移方程得到，需要直接给出？dp[0]=0，dp[1]=1
计算顺序计算 dp[i] 时，它依赖的状态是否都已算好？i 从 2 递增到 n
答案位置答案是哪个状态？dp[n]

其中状态定义最关键，要写成完整的句子，例如“dp[i] 表示以 nums[i] 结尾的子数组的最大和”。状态的含义写清楚了，转移方程和初始值通常也就能推出来。

509 斐波那契数：三种写法

写法一：暴力递归。 直接按定义写。画出 fib(5) 的递归树，可以看到 f(3) 被计算了 2 次，f(2) 被计算了 3 次：

fib(5) 的暴力递归树共 15 次调用。蓝色是第一次算某个 f(k)，橙色是重复再算一遍的 f(k)（f(3) 算了 2 次，f(2) 算了 3 次）

调用次数满足 calls(n) = calls(n−1) + calls(n−2) + 1：fib(5) 调用 15 次，fib(20) 调用 21891 次。调用次数与斐波那契数本身同阶增长，是指数级的。

写法二：记忆化搜索（自顶向下）。 保留递归的写法，增加一个备忘录 memo，已经算过的值直接返回。这样每个 f(k) 只真正计算一次，共有 n+1 个状态，时间 O(n)，空间 O(n)（备忘录加递归栈）。

写法三：递推（自底向上）。 按照上面的五个步骤，从小到大依次填表。由于 dp[i] 只依赖前两项，只需保留两个变量，空间可以降到 O(1)。

输入 n ≥ 0，输出第 n 个斐波那契数，其中 F(0)=0，F(1)=1。prev 和 cur 保存相邻的两项：在要计算第 i 项的那一轮开始时，prev = F(i−2)，cur = F(i−1)。算出 next = prev + cur 后，令 prev = cur，cur = next，两者整体向后移动一项。i 走到 n 后循环结束，cur 就是 F(n)。

long long fibIter(int n) {
    if (n < 2) return n;
    long long prev = 0, cur = 1;                       // F(i-1), F(i)
    for (int i = 2; i <= n; ++i) {

        prev = cur; cur = next;

def fib_iter(n):

    prev, cur = 0, 1                                          # F(i-1), F(i)；Python int 不溢出，不需要 long long

怎样想出状态和转移

上面的五个步骤说明了动态规划要写出哪些东西，却没有说明状态从哪里来。面对一道新题，可以按下面几条经验去找。

第一，先写暴力递归，递归函数的参数就是状态。 把问题看成一连串的决定，写一个递归函数尝试每一步的所有选择。为了把“剩下的事”做完，函数需要知道哪些信息，这些信息就是它的参数，也就是状态。画出递归树，如果同样的参数反复出现（如 fib(3) 被计算了两次），就加上备忘录，这已经是动态规划了。

第二，盯住最后一步。 问：最优解（或者某一个方案）的最后一步是什么？按最后一步的所有可能性分类，去掉最后一步后，剩下的部分就是规模更小的同类问题。爬楼梯的最后一步是走 1 级或 2 级；背包问题看最后一件物品拿还是不拿；两个字符串的问题看两个末尾字符是否相同；编辑距离看最后一次操作是替换、删除还是插入。分类必须不重复、不遗漏：求方案数时把各类相加，求最值时取各类中最好的一个，判断可行性时只要有一类可行即可。

第三，状态要做到只看现在、不问过去。 知道了状态，以后怎样走就只由状态决定，与之前怎样走到这里无关，这叫作无后效性。如果写转移时发现还需要知道更多的历史，说明状态不够，就把这部分历史加进状态。例如股票题只记“第 i 天”不够，因为今天能不能卖出，取决于手中是否持有股票，所以状态是（第 i 天，是否持有）；有冷冻期时，还要再加上“昨天是否刚卖出”（11.11）。

第四，先试常见的状态形状。 大多数题的状态是下面几种形状之一：一个数组时，用“前 i 个元素”，连续子数组用“以第 i 个元素结尾”（11.5）；两个字符串时，用两个前缀长度 (i, j)（11.7）；回文和区间合并时，用区间 [i, j]（11.10）；选物品且有容量限制时，用（前 i 件物品，容量 c）（11.8）；需要额外记住少量信息时，再加一个小维度，如持有与否、已用次数、位掩码（11.11、11.13）。依次尝试这些形状，通常就能找到合适的状态。

第五，用数据范围检验状态是否可行。 总时间约等于状态数乘以每个状态的转移代价，代入数据范围，与每秒约 10⁸ 次运算比较（7.2）。n ≤ 5000 时可以用 dp[n][n]；n ≤ 10⁵ 时只能用 dp[n]，并且每次转移是 O(1) 或 O(log n)；某个数值（容量、金额）不超过 10⁴ 左右时，它本身可以作为一维状态。估算超时，就说明状态选得太细，要换一种定义。

第六，拿不准计算顺序时，先写记忆化搜索。 记忆化搜索按需递归，不用考虑先算哪个状态；写对之后，如果递归太深或需要压缩空间，再按状态之间的依赖关系改成递推。

下面用一道没讲过的题完整演示这几条经验。

例：91 解码方法。 字母 A～Z 分别编码为 1～26，给出一串数字，问有多少种方法把它解码回字母。例如 "226" 可以解码为 2 2 6、22 6、2 26，共 3 种；"06" 无法解码，因为 0 不能单独解码，06 也不是合法的编码。[4]

暴力的想法是从左到右逐个决定：下一个字母由 1 位数字还是 2 位数字解码。看最后一步：解码出的最后一个字母，要么来自最后 1 位数字（这一位不是 0），要么来自最后 2 位数字（这两位组成的数在 10 到 26 之间）。去掉最后一个字母后，剩下的是一个更短的前缀，于是状态是 dp[i]：字符串的前 i 个字符有几种解码方法。两类情况互不重叠，方案数相加：

dp[i] = （s[i−1] ≠ '0' 时）dp[i−1] + （s[i−2..i−1] 在 10～26 之间时）dp[i−2]

初始值 dp[0] = 1，表示空串只有一种解码方法（什么也不做），这样“整个串恰好由一个字母解码”的情况也能通过 dp[i−1] 或 dp[i−2] 计入。答案是 dp[n]。状态有 n+1 个，每次转移 O(1)，总时间 O(n)。

long long numDecodings(const string& s) {
    int n = (int)s.size();
    vector<long long> dp(n + 1, 0);   // dp[i]：前 i 个字符的解码方法数
    dp[0] = 1;                         // 空串有一种解码方法
    for (int i = 1; i <= n; ++i) {
        if (s[i - 1] != '0') dp[i] += dp[i - 1];          // 最后一个字母来自 1 位数字
        if (i >= 2) {
            int two = (s[i - 2] - '0') * 10 + (s[i - 1] - '0');
            if (two >= 10 && two <= 26) dp[i] += dp[i - 2]; // 最后一个字母来自 2 位数字
        }
    }
    return dp[n];
}
// numDecodings("226") == 3，numDecodings("06") == 0，numDecodings("2101") == 1
def num_decodings(s):
    n = len(s)
    dp = [0] * (n + 1)          # dp[i]：前 i 个字符的解码方法数
    dp[0] = 1                   # 空串有一种解码方法
    for i in range(1, n + 1):
        if s[i - 1] != '0':
            dp[i] += dp[i - 1]  # 最后一个字母来自 1 位数字
        if i >= 2 and 10 <= int(s[i - 2:i]) <= 26:
            dp[i] += dp[i - 2]  # 最后一个字母来自 2 位数字
    return dp[n]

手算 "226"：dp[0] = 1；i=1，'2' 不是 0，dp[1] = dp[0] = 1；i=2，'2' 不是 0，加 dp[1]，"22" 在 10～26 之间，再加 dp[0]，dp[2] = 2；i=3，'6' 不是 0，加 dp[2]，"26" 在范围内，再加 dp[1]，dp[3] = 2 + 1 = 3。对 "2101"：前两位只能解码为 21 或 2 1，但第三位 0 只能与前面的 1 组成 10，所以前两位必须是单独的 2，最终只有 2、10、1 这一种。

小结： 动态规划适用于同时具有最优子结构（大问题的最优解由子问题的最优解组成）和重叠子问题（同一个子问题被多次用到）的问题。分治算法（如归并排序）的子问题互不重叠，所以不需要备忘录；动态规划的子问题有重叠，所以要把结果存在表中。记忆化搜索和递推是同一个动态规划的两种实现：前者从目标出发向下拆分，后者从初始值出发向上填表。

int stepsToOne_4_2(int n) {
    vector<int> dp(n + 1, 0);                          // dp[1] = 0
    for (int i = 2; i <= n; ++i) {
        dp[i] = dp[i - 1] + 1;                         // 减 1
        if (i % 2 == 0) dp[i] = min(dp[i], dp[i / 2] + 1);
        if (i % 3 == 0) dp[i] = min(dp[i], dp[i / 3] + 1);

}

long long bracketsDepth_4_3(int n, int d) {
    vector<long long> dp(d + 1, 0);                    // dp[j]：当前长度下深度为 j 的方案数
    dp[0] = 1;                                         // 空串
    for (int step = 0; step < 2 * n; ++step) {
        vector<long long> nd(d + 1, 0);
        for (int j = 0; j <= d; ++j) {
            if (dp[j] == 0) continue;
            if (j + 1 <= d) nd[j + 1] += dp[j];        // 写 '('
            if (j - 1 >= 0) nd[j - 1] += dp[j];        // 写 ')'
        }
        dp = nd;
    }
    return dp[0];
}

11.5 一维动态规划：爬楼梯、打家劫舍、最大子数组和、乘积最大子数组

一维动态规划的状态只有一个下标 i，是最简单的一类动态规划。题目给出一个数组或一个整数 n，答案只与走到第 i 步时的情况有关，每一步只有两三种决策，例如走 1 级还是 2 级、偷还是不偷、接在前一段后面还是重新开始。本节四道题的状态有两种定义方式：前 i 个元素范围内的最优值（70、198）和以第 i 个元素结尾的最优值（53、152）。区分这两种定义，是后面学习子序列动态规划的基础。

这类题的计算过程都可以写成下面的形式：

dp[基例] = ...
for i in 递增顺序:
    dp[i] = 合并( dp[i-1] 经过选择 A,  dp[i-2] 经过选择 B, ... )
answer = dp[n] 或 max(dp[0..n-1])

70 爬楼梯

每次可以走 1 级或 2 级台阶，求走到第 n 级（n ≥ 1）的不同方法数。设 dp[i] 为走到第 i 级的方法数。走到第 i 级的最后一步，要么是从第 i−1 级走 1 级上来，要么是从第 i−2 级走 2 级上来，这两类方法互不重叠，所以 dp[i] = dp[i−1] + dp[i−2]。从小到大计算，算 dp[i] 时 dp[1..i−1] 都已算好，答案是 dp[n]。

int climbStairs(int n) {
    if (n <= 2) return n;
    vector<int> dp(n + 1);
    dp[1] = 1; dp[2] = 2;
    for (int i = 3; i <= n; ++i) dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}
def climb_stairs(n):
    if n <= 2: return n
    dp = [0] * (n + 1)
    dp[1] = 1; dp[2] = 2
    for i in range(3, n + 1): dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]

198 打家劫舍

输入非负整数数组 nums，nums[i] 是第 i 间房中的钱，不能偷相邻的两间房，求能偷到的最大金额。设 dp[i] 为只考虑下标 0..i 的房子时能偷到的最大金额。对第 i 间房只有两种选择：不偷，结果是 dp[i−1]；偷，那么第 i−1 间不能偷，结果是 dp[i−2] + nums[i]。dp[i] 取两者中较大的一个，答案是 dp[n−1]。

int rob(const vector<int>& nums) {
    int n = nums.size();
    if (n == 1) return nums[0];
    vector<int> dp(n);
    dp[0] = nums[0]; dp[1] = max(nums[0], nums[1]);
    for (int i = 2; i < n; ++i) dp[i] = max(dp[i - 1], dp[i - 2] + nums[i]);
    return dp[n - 1];
}
def rob(nums):
    n = len(nums)
    if n == 1: return nums[0]
    dp = [0] * n
    dp[0] = nums[0]; dp[1] = max(nums[0], nums[1])
    for i in range(2, n): dp[i] = max(dp[i - 1], dp[i - 2] + nums[i])
    return dp[n - 1]

时间 O(n)，空间 O(n)。由于 dp[i] 只用到 dp[i−1] 和 dp[i−2]，可以改成两个变量滚动，空间降到 O(1)。

53 最大子数组和

输入非空整数数组，求和最大的连续子数组的和。设 cur 为以当前元素结尾的连续子数组的最大和，best 为目前为止所有 cur 中的最大值。以 nums[i] 结尾的子数组，要么只含 nums[i] 自己，要么接在以 nums[i−1] 结尾的最优子数组之后，所以 cur = max(nums[i], cur + nums[i])，然后用 cur 更新 best。扫描完毕后返回 best。

状态之所以定义为以 i 结尾，是因为子数组要求连续：只有知道前一段恰好在 i−1 处结束，才能把 nums[i] 接上去。最优子数组可能在任何位置结尾，所以答案是所有 cur 中的最大值，而不是最后一个 cur。

int maxSubArray(const vector<int>& nums) {
    int cur = nums[0], best = nums[0];                 // cur：以当前元素结尾的最大和
    for (int i = 1; i < (int)nums.size(); ++i) {
        cur = max(nums[i], cur + nums[i]);
        best = max(best, cur);
    }
    return best;
}
def max_sub_array(nums):
    cur = best = nums[0]                                      # cur：以当前元素结尾的最大和
    for i in range(1, len(nums)):
        cur = max(nums[i], cur + nums[i])
        best = max(best, cur)
    return best

时间 O(n)，空间 O(1)。这种写法称为 Kadane 算法。

152 乘积最大子数组

输入非空整数数组，其中可能有负数和 0，求乘积最大的连续子数组的乘积。与 53 题不同的是，一个负数乘以负数会变成正数：以 nums[i−1] 结尾的最小乘积（一个绝对值很大的负数）乘上负数 x 后，可能变成最大的乘积。所以要同时维护以当前元素结尾的最大乘积 mx 和最小乘积 mn。以 x = nums[i] 结尾的乘积只有三种来源：x 自己、mx·x、mn·x，三者的最大值作为新的 mx，最小值作为新的 mn，再用 mx 更新全局最大值 best。

maxProduct([-2,3,-4])：mn 保存“最负”的乘积

第1步 · i=0：mx=mn=best=−2

mx=−2  mn=−2  best=−2
2 / 3

第2步 · i=1，x=3：候选 {3, −2×3, −2×3} → mx=3，mn=−6

mx=3  mn=−6  best=3
3 / 3

第3步 · i=2，x=−4：候选 {−4, 3×(−4)=−12, (−6)×(−4)=24} → mx=24

mx=24  mn=−12  best=24：最负的 −6 遇到负数翻身成最大
int maxProduct(const vector<int>& nums) {
    int mx = nums[0], mn = nums[0], best = nums[0];
    for (int i = 1; i < (int)nums.size(); ++i) {
        int x = nums[i];
        int a = mx * x, b = mn * x;                    // 先用旧值算出两个候选
        mx = max({x, a, b});
        mn = min({x, a, b});
        best = max(best, mx);
    }
    return best;
}
def max_product(nums):
    mx = mn = best = nums[0]
    for i in range(1, len(nums)):
        x = nums[i]
        a, b = mx * x, mn * x                                 # 先用旧值算出两个候选；Python int 不溢出
        mx = max(x, a, b)                                     # 也可以一行同时赋值：mx, mn = max(x, mx*x, mn*x), min(x, mx*x, mn*x)，右边用的都是旧值
        mn = min(x, a, b)
        best = max(best, mx)
    return best

时间 O(n)，空间 O(1)。

小结： 状态定义为前 i 个元素范围内的最优值时，答案在 dp[n] 或 dp[n−1]；定义为以 i 结尾的最优值时，答案是所有 dp[i] 中的最大值。连续子数组问题的状态几乎都定义为以 i 结尾。

int gymCost_5_1(const vector<int>& cost) {

int robRange_5_2(const vector<int>& a, int l, int r) {   // 只看下标 [l, r] 这一段的打家劫舍
    int prev = 0, cur = 0;                             // 滚动的 dp[i-2]、dp[i-1]

        int nxt = max(cur, prev + a[i]);

    }
    return cur;
}

练习 11.5-3【变形】数字暗号有几种译法

题面：暗号表 A→1，B→2，…，Z→26。给一串数字 s，问有多少种方法把它译回字母串。'0' 不能单独译，也不能作两位数的首位（"06" 不能译成 F）。1 ≤ |s| ≤ 40。样例："226" → 3（BBF、BZ、VF）；"06" → 0；"11106" → 2（AAJF、KJF）；"10" → 1。

提示

状态：dp[i] 表示前 i 个字符的译法数，dp[0] 对应空前缀。转移：最后一个字母要么由最后 1 个字符译出（要求它不是 '0'），要么由最后 2 个字符译出（要求组成 10～26）；两类译法的最后一个字母不同，互不重叠，满足条件的就加上。和 70 题相比："走 1 级 / 走 2 级"各自多了一个能不能走的条件。相近题型：LeetCode 91。

参考答案
int decodeWays_5_3(const string& s) {
    int n = s.size();
    vector<int> dp(n + 1, 0);
    dp[0] = 1;                                         // 空前缀：1 种（什么都不译）
    for (int i = 1; i <= n; ++i) {
        if (s[i - 1] != '0') dp[i] += dp[i - 1];       // 最后 1 个字符单独译
        if (i >= 2 && s[i - 2] != '0') {
            int two = (s[i - 2] - '0') * 10 + (s[i - 1] - '0');
            if (two <= 26) dp[i] += dp[i - 2];         // 最后 2 个字符合起来译
        }
    }
    return dp[n];
}
def decode_ways_5_3(s):
    n = len(s)
    dp = [0] * (n + 1)
    dp[0] = 1                                                 # 空前缀：1 种（什么都不译）
    for i in range(1, n + 1):
        if s[i - 1] != '0': dp[i] += dp[i - 1]                # 最后 1 个字符单独译
        if i >= 2 and s[i - 2] != '0':
            two = int(s[i - 2:i])
            if two <= 26: dp[i] += dp[i - 2]                  # 最后 2 个字符合起来译
    return dp[n]

练习 11.5-4【变形】拿分并作废相邻值

题面：一堆数字卡 a。每次拿走一张值为 x 的卡得 x 分，同时所有值为 x−1 和 x+1 的卡作废（直接丢掉、不得分）。卡拿完或作废完为止，求最大总分。1 ≤ n ≤ 2×10^4，1 ≤ a[i] ≤ 10^4。样例：[3,4,2] → 6（拿 4 作废 3，再拿 2）；[2,2,3,3,3,4] → 9（三张 3）；[1] → 1。

提示

先换个看法：值为 x 的卡要拿就全拿，得分 sum[x] = x × 出现次数；拿了 x 就不能拿 x−1 和 x+1。于是问题变成对数组 sum[1..max] 做打家劫舍——"相邻的房子"变成"相邻的值"。状态：dp[x] 表示只考虑值 1..x 时的最大得分；转移：值 x 拿或不拿。相近题型：LeetCode 740。

参考答案
int pickPoints_5_4(const vector<int>& a) {
    int mx = *max_element(a.begin(), a.end());
    vector<int> sum(mx + 1, 0);                        // sum[x]：所有值为 x 的卡加起来
    for (int x : a) sum[x] += x;
    int prev = 0, cur = 0;                             // 对 sum 做打家劫舍
    for (int x = 1; x <= mx; ++x) {
        int nxt = max(cur, prev + sum[x]);
        prev = cur; cur = nxt;
    }
    return cur;
}

    total = [0] * (mx + 1)                                    # total[x]：所有值为 x 的卡加起来（sum 是内置函数名，换个名字）
    for x in a: total[x] += x
    prev, cur = 0, 0                                          # 对 total 做打家劫舍

    return cur

练习 11.5-5【综合】环形跑道的最大得分段

题面：环形跑道分成 n 段，第 i 段得分 a[i]（可正可负），首尾相连。选一段连续、非空、长度不超过 n 的跑道（可以跨过终点回到起点），求得分和的最大值。1 ≤ n ≤ 3×10^4，−1000 ≤ a[i] ≤ 1000。样例：[1,−2,3,−2] → 3；[5,−3,5] → 10（跨尾的 5,5）；[−3,−2,−3] → −2。

提示

分两种情况。不跨尾：就是 53 的 Kadane，状态是"以 i 结尾的最大和"。跨尾：选中的部分 = 总和 − 中间一段没选的，要让没选的那段和最小，于是再用一次 Kadane 求"以 i 结尾的最小和"。两种取大。边界：全是负数时，"中间最小段"就是整个数组，总和减它得到 0，对应的是空子数组，不合法，此时只能返回不跨尾的答案。对拍：枚举起点和长度 1..n 的 O(n²) 暴力。相近题型：LeetCode 918。

参考答案
int ringMaxSum_5_5(const vector<int>& a) {
    int total = 0, curMax = 0, bestMax = a[0], curMin = 0, bestMin = a[0];
    for (int x : a) {
        total += x;
        curMax = max(x, curMax + x); bestMax = max(bestMax, curMax);   // 不跨尾：普通 Kadane
        curMin = min(x, curMin + x); bestMin = min(bestMin, curMin);   // 跨尾：找中间和最小的一段
    }
    if (bestMax < 0) return bestMax;                   // 全是负数：不能把整个数组都"不选"
    return max(bestMax, total - bestMin);
}
def ring_max_sum_5_5(a):
    total, cur_max, best_max, cur_min, best_min = 0, 0, a[0], 0, a[0]
    for x in a:
        total += x
        cur_max = max(x, cur_max + x); best_max = max(best_max, cur_max)   # 不跨尾：普通 Kadane
        cur_min = min(x, cur_min + x); best_min = min(best_min, cur_min)   # 跨尾：找中间和最小的一段
    if best_max < 0: return best_max                          # 全是负数：不能把整个数组都"不选"
    return max(best_max, total - best_min)
