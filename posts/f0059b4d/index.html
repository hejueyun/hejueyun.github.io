6.5 层序遍历：用队列逐层向外扩展 必读

有些题目关心节点在第几层，例如每层的最大值、第 k 层有几个节点、离根最近的叶子。DFS 一直向下走到底，不同层的节点交错着被访问；这类题目需要另一种顺序：先把第 1 层全部处理完，再处理第 2 层。

队列使先发现的节点先处理

BFS（Breadth-First Search，广度优先搜索）使用一个队列：处理一个节点时，把它的孩子排到队尾。同一层的节点都是由上一层的节点按顺序放入的，所以它们在队列中相邻，并且排在下一层的节点之前。

每一轮循环固定做四件事：读取队首（front），删除队首（pop），处理这个节点，把它的孩子入队（push）。

vector<int> bfsOrder(TreeNode* root) {
    vector<int> out;
    if (root == nullptr) return out;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        TreeNode* cur = q.front();     // 1. 看队首
        q.pop();                       // 2. 删掉队首
        out.push_back(cur->val);       // 3. 处理它
        if (cur->left) q.push(cur->left);     // 4. 孩子排到队尾
        if (cur->right) q.push(cur->right);
    }
    return out;
}
from collections import deque

def bfs_order(root):
    out = []
    if root is None: return out
    q = deque()
    q.append(root)
    while q:
        cur = q.popleft()               # 1+2. 看队首并删掉：popleft() 会把它返回
        out.append(cur.val)             # 3. 处理它
        if cur.left: q.append(cur.left)     # 4. 孩子排到队尾
        if cur.right: q.append(cur.right)
    return out

逐行说明：第 3 行处理空树，否则第 5 行会把 nullptr 入队，第 9 行的 cur->val 就会崩溃；第 7–8 行先 front 再 pop，因为 pop() 不返回被删除的元素；第 10–11 行的 if (cur->left) 等价于 if (cur->left != nullptr)，只让真实节点入队。

vector<vector<int>> levelGroups(TreeNode* root) {
    vector<vector<int>> res;
    if (root == nullptr) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int sz = (int)q.size();        // 这一层有几个节点，先记下来
        vector<int> level;
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            level.push_back(cur->val);
            if (cur->left) q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
        res.push_back(level);
    }
    return res;
}
def level_groups(root):
    res = []
    if root is None: return res
    q = deque()
    q.append(root)
    while q:
        sz = len(q)                     # 这一层有几个节点，先记下来
        level = []
        for i in range(sz):             # 写成 range(len(q)) 也对：range 在开头就算好了次数
            cur = q.popleft()
            level.append(cur.val)
            if cur.left: q.append(cur.left)
            if cur.right: q.append(cur.right)
        res.append(level)
    return res

为什么要先把 sz 存起来？ 如果写成 for (int i = 0; i < (int)q.size(); ++i)，循环中一边出队一边入队，q.size() 在不断变化。第 2 轮开始时队列是 [2,3]：i=0 时出队 2、入队 4，size 仍为 2；i=1 时出队 3、入队 5 和 6，size 变成 3，条件 2 < 3 成立，于是第 3 层的 4 也被算进了第 2 层。实际输出会变成 [[1],[2,3,4],[5,6]]。

要点

  1. BFS 使用队列，DFS 使用栈（递归时使用的是系统的调用栈）。
  2. 队列先进先出，所以节点按层出队，每层内部从左到右。
  3. 按层分组时，每轮先记下 sz = q.size()，再出队 sz 个节点。
  4. 每个节点入队、出队各一次，时间 O(n)；队列中最多同时存放一层，空间等于最宽一层的节点数。

练一练 1：第 k 层有几个节点

题面：根在第 1 层，返回第 k 层的节点个数；树没有那么多层时返回 0。

提示

每轮开始时的 sz 就是这一层的节点数，数到第 k 轮时直接返回 sz。

参考答案
int countLevelK(TreeNode* root, int k) {
    if (root == nullptr || k < 1) return 0;
    queue<TreeNode*> q;
    q.push(root);
    int depth = 1;
    while (!q.empty()) {
        int sz = (int)q.size();
        if (depth == k) return sz;     // 队列里正好是第 k 层
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            if (cur->left) q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
        ++depth;
    }
    return 0;                          // 树没有这么多层
}
def count_level_k(root, k):
    if root is None or k < 1: return 0
    q = deque()
    q.append(root)
    depth = 1
    while q:
        sz = len(q)
        if depth == k: return sz        # 队列里正好是第 k 层
        for i in range(sz):
            cur = q.popleft()
            if cur.left: q.append(cur.left)
            if cur.right: q.append(cur.right)
        depth += 1
    return 0                            # 树没有这么多层

对 [1,2,3,4,null,5,6]，k=1、2、3、4 时分别返回 1、2、3、0。与例题的区别： 多了层号 depth，并且不需要记录节点的值，到第 k 层就提前返回。

练一练 2：每层节点值之和

题面：返回每一层的节点值之和。[1,2,3,4,null,5,6] 应返回 [1, 5, 15]。

vector<long long> levelSums(TreeNode* root) {
    vector<long long> res;
    if (root == nullptr) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int sz = (int)q.size();
        long long s = 0;
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            s += cur->val;
            if (cur->left) q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
        res.push_back(s);
    }
    return res;
}
def level_sums(root):
    res = []
    if root is None: return res
    q = deque()
    q.append(root)

        sz = len(q)
        s = 0                           # Python 整数不会溢出，不需要 long long

与例题的区别： 把 vector<int> level 换成一个累加器 s。在同样的结构中把 s 换成最大值，或者换成本层最后一个节点，就得到每层最大值、右视图等题目。

练习 6.5-1【基础】每个职级的最高绩效

题面：公司的汇报关系是一棵二叉树，同一层的人属于同一个职级。节点值是绩效分（可能为负）。写 vector<int> levelMaxes(TreeNode* root)，返回每一层的最大值，从第 1 层开始。节点数不超过 10⁴。

样例：[4,9,2,-3,7,null,8] → {4, 9, 8}；[-4,-7,-2] → {-4, -2}；空树 → {}。

提示

"每一层"→ 按层分组的 BFS，每轮开头记 sz。和练一练 2（每层之和）相比只改累加器：s 换成 best，s += val 换成 best = max(best, val)。best 的初值要用 INT_MIN，写成 0 会在全负的层上出错。相近题型：LeetCode 515。

参考答案
vector<int> levelMaxes(TreeNode* root) {
    vector<int> res;
    if (root == nullptr) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int sz = (int)q.size();
        int best = INT_MIN;                 // 初值用 INT_MIN：全负的层也能比对
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            best = max(best, cur->val);
            if (cur->left) q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
        res.push_back(best);
    }
    return res;
}
def level_maxes(root):

            best = max(best, cur.val)

练习 6.5-2【变形】站在楼的右边能看到谁

题面：把二叉树想成一栋楼，每层是一个楼层。站在楼的右侧往左看，每层只能看到最右边的那个节点。写 vector<int> rightView(TreeNode* root)，从上到下返回能看到的值。节点数不超过 10⁴。

样例：[6,2,8,null,4,7] → {6, 8, 7}；[1,2,3,4] → {1, 3, 4}（第 3 层只有左边的 4，它也能被看到）。

提示

还是按层分组的 BFS。改动：不收集整层，只在 i == sz − 1（本层最后一个出队的节点）时记录。常见错误是"一直往右走"：从根沿 right 指针走到底，样例 2 中 3 没有孩子，会漏掉第 3 层的 4。相近题型：LeetCode 199。

参考答案
vector<int> rightView(TreeNode* root) {
    vector<int> res;
    if (root == nullptr) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int sz = (int)q.size();
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            if (i == sz - 1) res.push_back(cur->val);   // 本层最后一个出队的节点
            if (cur->left) q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
    }
    return res;
}
def right_view(root):
    res = []
    if root is None: return res
    q = deque()
    q.append(root)
    while q:
        sz = len(q)
        for i in range(sz):
            cur = q.popleft()
            if i == sz - 1: res.append(cur.val)         # 本层最后一个出队的节点
            if cur.left: q.append(cur.left)
            if cur.right: q.append(cur.right)
    return res

练习 6.5-3【变形】蛇形巡检：一层从左到右，下一层从右到左

题面：巡检员按层检查一棵树，第 1 层从左往右，第 2 层从右往左，第 3 层又从左往右，交替进行。写 vector<vector<int>> zigzagLevels(TreeNode* root)，返回每层的巡检顺序。节点数不超过 10⁴。

提示

vector<vector<int>> zigzagLevels(TreeNode* root) {
    vector<vector<int>> res;
    if (root == nullptr) return res;
    queue<TreeNode*> q;
    q.push(root);
    bool leftToRight = true;
    while (!q.empty()) {
        int sz = (int)q.size();
        vector<int> level;
        for (int i = 0; i < sz; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            level.push_back(cur->val);
            if (cur->left) q.push(cur->left);       // 入队顺序不变：仍是先左后右
            if (cur->right) q.push(cur->right);
        }
        if (!leftToRight) reverse(level.begin(), level.end());   // 只翻转这一层的结果
        res.push_back(level);
        leftToRight = !leftToRight;
    }
    return res;
}

题面：写 int minDepthBfs(TreeNode* root)，返回离根最近的叶子所在的层号（根在第 1 层）；空树返回 0。节点数不超过 10⁵。

提示

"最近"→ BFS：按层往外推，第一次遇到叶子时的层号就是答案，可以立刻返回，不用把整棵树走完。陷阱在"叶子"的定义：两个孩子都为空才是叶子。如果写成分解型 min(左高, 右高) + 1，样例 2 中空的左子树会返回 0，结果变成 1，这是错的。practice 里用一个正确处理了"只有一个孩子"的递归版本对拍。相近题型：LeetCode 111。

int minDepthBfs(TreeNode* root) {

练习 6.5-5【综合】座位有没有"空洞"：完全二叉树判定

题面：礼堂座位按层排，每排从左往右坐。一棵树是完全二叉树，当且仅当按层序（含空位）写出来时，所有真实节点都排在第一个空位之前，也就是中间没有空洞。写 bool isCompleteTree(TreeNode* root)。空树算是。节点数不超过 10⁴。

样例：[1,2,3,4,5,6] → true；[1,2,3,4,null,6] → false（4 和 6 之间空了一个位置）；[1,null,2] → false；[1,2] → true；[] → true。

提示

和 toLevelArray 一样，空孩子也入队，这样队列的出队顺序就是"含空位的层序"。用一个 bool 记"是否已经见过空位"：见到空位就置 true；之后只要再出队一个真实节点，就说明中间有洞，返回 false。边界：空树时队列里只有一个 nullptr，循环结束返回 true。practice 用另一种思路对拍：给节点按 1、2、3… 的堆编号（左孩子 2i，右孩子 2i+1），完全二叉树当且仅当最大编号等于节点数。相近题型：LeetCode 958。

参考答案
bool isCompleteTree(TreeNode* root) {
    queue<TreeNode*> q;
    q.push(root);
    bool seenNull = false;              // 是否已经见过空位
    while (!q.empty()) {
        TreeNode* cur = q.front();
        q.pop();
        if (cur == nullptr) { seenNull = true; continue; }
        if (seenNull) return false;     // 空位后面又出现真实节点：中间有洞
        q.push(cur->left);              // 和 toLevelArray 一样，空孩子也入队
        q.push(cur->right);
    }
    return true;
}
def is_complete_tree(root):
    q = deque()
    q.append(root)
    seen_null = False                   # 是否已经见过空位
    while q:
        cur = q.popleft()
        if cur is None: seen_null = True; continue
        if seen_null: return False      # 空位后面又出现真实节点：中间有洞
        q.append(cur.left)              # 和 toLevelArray 一样，空孩子也入队
        q.append(cur.right)
    return True

易错点

  • 用 q.size() 直接当 for 的循环条件，层与层会混在一起。
  • 以为 q.pop() 会返回队首；要先 front() 读出来再 pop()。
  • 把 nullptr 入队后又直接访问 ->val：要么入队前判空，要么出队后判空，二选一。

小结： BFS 使用队列：取出队首、弹出、处理、把孩子入队；需要分层时，在每一轮开始时记下当前队列的长度 sz。

6.6 树上递归的两种写法：遍历型与分解型 必读

面试中的树题形式多样：节点个数、高度、叶子数、节点值之和、最大值、是否存在某个值……它们都可以用同一个思路写出来，而且只有两种写法。掌握这两种写法，遇到新题时就知道从哪里入手。

两种写法

  • 遍历型：把树走一遍（通常在前序位置处理），用一个外部变量记录答案，走到每个节点时更新它，函数本身返回 void。这就像拿着计数器从楼顶走到楼底，每见到一个人按一下。
  • 分解型：函数的返回值就是以 node 为根的子树的答案，当前节点的答案由左子树和右子树的答案得出。这就像每层楼的负责人只向上一层汇报本层及以下共有几个人。

写分解型之前，先用一句话确定函数的含义，然后回答三个问题：

  1. 函数的含义：f(node) 返回什么？这个含义必须对以任意节点为根的子树都成立。
  2. 空树返回什么？这是递归的出口。
  3. 已知 f(node->left) 和 f(node->right)，怎样计算 f(node)？

逐步执行：用分解型统计节点数

函数的含义是：countNodes(node) 返回以 node 为根的子树中的节点个数。空树有 0 个；否则是左子树的个数加右子树的个数，再加上自己 1 个。

void heightWalk(TreeNode* node, int depth, int& best) {
    if (node == nullptr) return;
    best = max(best, depth);                      // depth：node 在第几层
    heightWalk(node->left, depth + 1, best);
    heightWalk(node->right, depth + 1, best);
}
int heightByWalk(TreeNode* root) { int best = 0; heightWalk(root, 1, best); return best; }
def height_walk(node, depth, best):              # depth 是 int，重新赋值影响不到调用者，相当于按值传
    if node is None: return
    best[0] = max(best[0], depth)                 # best = [0] 是只有一个元素的 list，所有层改同一个对象，对应 int&
    height_walk(node.left, depth + 1, best)
    height_walk(node.right, depth + 1, best)
def height_by_walk(root): best = [0]; height_walk(root, 1, best); return best[0]

逐行说明：depth 按值传递，每层有自己的副本，从父节点得到 depth，向孩子传 depth + 1，回到父节点时父节点的 depth 没有被改动；best 按引用传递（int&），所有层修改的是同一个变量，这就是外部变量。入口函数中根在第 1 层，所以传入 1；空树时 best 保持为 0。

如何选择

  • 题目要求的量能由左右子树的同一个量得出（个数、高度、和、是否平衡）时，优先使用分解型，代码最短，也没有需要重置的外部变量。
  • 节点需要从上方传下来的信息（深度、路径、到根的和），或者答案是在遍历过程中顺便统计的量时，使用遍历型，信息通过参数向下传递。
  • 很多中等难度的题目两者混用：函数把子树的答案返回给父节点使用，同时在外部变量中更新全局答案，第 10 章的直径、最大路径和就是这样。

bool isSameTree(TreeNode* a, TreeNode* b) {

题面：一棵树是平衡的，当且仅当每个节点的左右子树高度差都不超过 1。写 bool isBalancedTree(TreeNode* root)，要求 O(n)。空树是平衡的。节点数不超过 10⁵。

样例：[6,2,8,null,null,7,9] → true；[1,2,null,3] → false（根的左边高 2、右边高 0）；[1,2,3,4,null,null,5,6,null,null,7] → false（根的左右都高 3，但节点 2 的左边高 2、右边高 0）；[] → true。

提示

分解型求高度，再加一个"坏了"的信号：函数 balancedHeight(node) 在子树平衡时返回高度，不平衡时返回 −1。合并时三步：左边是 −1 直接返回 −1；右边是 −1 直接返回 −1；两边高度差大于 1 返回 −1；否则返回 max(L, R) + 1。只检查根的左右高度是常见错误，样例 3 就是反例。在每个节点上都调用一次 height 也能做对，但是 O(n²)；practice 里正是用这个慢版本对拍。相近题型：LeetCode 110。

参考答案
int balancedHeight(TreeNode* node) {
    if (node == nullptr) return 0;
    int L = balancedHeight(node->left);
    if (L == -1) return -1;                   // 左子树里已经有不平衡的节点
    int R = balancedHeight(node->right);
    if (R == -1) return -1;
    if (abs(L - R) > 1) return -1;            // 自己这里不平衡
    return max(L, R) + 1;                     // 平衡时返回真实高度
}
bool isBalancedTree(TreeNode* root) { return balancedHeight(root) != -1; }
def balanced_height(node):

    L = balanced_height(node.left)
    if L == -1: return -1                     # 左子树里已经有不平衡的节点
    R = balanced_height(node.right)
    if R == -1: return -1
    if abs(L - R) > 1: return -1              # 自己这里不平衡
    return max(L, R) + 1                      # 平衡时返回真实高度

练习 6.6-5【综合】有几个部门的预算正好等于 target

题面：每个节点的值是这个人自己的预算（可正可负），一个"部门"是以某节点为根的整棵子树，部门预算是子树所有值之和。写 int countSubtreeSum(TreeNode* root, long long target)，返回部门预算恰好等于 target 的部门个数。节点数不超过 10⁵。

样例：[5,3,8,1,4]，target = 8 → 2（以 3 为根的 {3,1,4} 和以 8 为根的 {8}）；[0,1,-1]，target = 0 → 1（只有整棵树）；[0,0,0]，target = 0 → 3；空树 → 0。

提示

这是本节"怎样选"里说的混用：函数 subSumWalk(node) 的返回值是"子树之和"（分解型，交给父亲用），同时在函数里顺手检查"这个和是否等于 target"，命中就让引用参数 cnt 加 1（遍历型记全局答案）。每个节点只算一次子树和，总共 O(n)。如果对每个节点再单独调用一次 sumTree，就是 O(n²)，practice 里用它对拍。值可以为负，所以不能"和已经超过 target 就剪枝"。

参考答案
long long subSumWalk(TreeNode* node, long long target, int& cnt) {
    if (node == nullptr) return 0;
    long long s = subSumWalk(node->left, target, cnt) + subSumWalk(node->right, target, cnt) + node->val;
    if (s == target) cnt++;                   // 返回值交给父亲，外部变量记全局答案
    return s;
}
int countSubtreeSum(TreeNode* root, long long target) { int cnt = 0; subSumWalk(root, target, cnt); return cnt; }
def sub_sum_walk(node, target, cnt):          # cnt = [0]，对应 int&
    if node is None: return 0
    s = sub_sum_walk(node.left, target, cnt) + sub_sum_walk(node.right, target, cnt) + node.val
    if s == target: cnt[0] += 1               # 返回值交给父亲，外部变量记全局答案
    return s
def count_subtree_sum(root, target): cnt = [0]; sub_sum_walk(root, target, cnt); return cnt[0]

易错点

  • 遍历型的外部变量写成全局变量，在同一个程序里测第二棵树时忘了清零，答案会叠加。用"入口函数里初始化 + 引用参数"可以避免。
  • 分解型没先写函数含义，合并时就会犹豫"要不要加 1""叶子算不算"。先用一句话写出含义，再写代码。
  • 最大值的出口返回 0，遇到全负数的树就错。

小结： 遍历型写法走遍整棵树，用外部变量记录结果，信息通过参数向下传递；分解型写法先确定函数的含义和空树的返回值，再用左右子树的答案得出自己的答案。

void collectPaths(TreeNode* node, vector<int>& path, vector<vector<int>>& res) {
    if (node == nullptr) return;
    path.push_back(node->val);                       // 走进 node：记下
    if (node->left == nullptr && node->right == nullptr) {
        res.push_back(path);                         // 叶子：复制一份当前路径
    } else {
        collectPaths(node->left, path, res);
        collectPaths(node->right, path, res);
    }
    path.pop_back();                                 // 离开 node：撤销
}

    return res;
}

为什么要 pop_back？ 去掉这一行，path 只增不减。对 [5,3,8,1,4] 的实际输出是：

[5,3,1]
[5,3,1,4]

原因：从 1 返回到 3 时，1 还留在 path 中，进入 4 时路线变成 5→3→1→4，但 1 和 4 是兄弟，并不在同一条路径上。改正的方法是在函数末尾 pop_back，进入时 push 几次，离开时就 pop 几次。少写 pop 的版本 collectPathsNoPop 输出的正是上面这三行错误的结果。

另一种写法：path 按值传递。 把参数写成 vector<int> path（没有 &），每一层得到的是父节点那一份的副本，孩子怎样修改都不会影响父节点，于是不需要 pop_back。代价是每层都要复制一次 path，树很深时更慢。面试中两种写法都可以，按引用传递加 pop_back 是更通用的回溯写法。

void collectPathsByValue(TreeNode* node, vector<int> path, vector<vector<int>>& res) {
    if (node == nullptr) return;
    path.push_back(node->val);
    if (node->left == nullptr && node->right == nullptr) { res.push_back(path); return; }
    collectPathsByValue(node->left, path, res);
    collectPathsByValue(node->right, path, res);
}
def collect_paths_by_value(node, path, res):        # Python 传 list 只传引用，没有"按值传"
    if node is None: return
    path = path + [node.val]                         # 造一个新 list 当本层的副本，父亲那份不受影响
    if node.left is None and node.right is None: res.append(path); return
    collect_paths_by_value(node.left, path, res)
    collect_paths_by_value(node.right, path, res)

要点

  1. path 表示从根到当前节点的路线；进入节点时 push，离开节点时 pop，这就是回溯中的做选择和撤销选择。
  2. 判断路径到头要看是否到达叶子（两个孩子都为空），而不是空节点：用空节点判断会使每条路径被记录两次（左边的空节点一次，右边的空节点一次）。
  3. 简单的数值（当前的和、当前的深度）按值传递即可，返回上一层时自动恢复；vector 这样的大对象按引用传递，并手动 pop。

void targetPathsDfs(TreeNode* node, long long remain, vector<int>& path, vector<vector<int>>& res) {
    if (node == nullptr) return;
    path.push_back(node->val);

练习 6.7-3【综合】没有重复站点的路线有几条

题面：节点值是 0 到 9 的站点编号，同一个编号可能出现在树的多个位置。一条根到叶路线如果经过的编号两两不同，就叫"无重复路线"。写 int distinctPathCount(TreeNode* root)，返回无重复路线的条数。节点数不超过 10⁵，空树返回 0。

样例：[1,2,1,3,2] → 1（1→2→3 无重复；1→2→2 和 1→1 都有重复）；[4,1,2,3,4,null,5] → 2（4→1→3、4→2→5）；[3,3,3] → 0；[5] → 1；[] → 0。

提示

path 回溯的另一种形式：不存 path 本身，存"path 上每个编号出现了几次"的计数数组 freq（10 个格子）。进门 freq[val]++，出门 freq[val]--，这就是 push / pop。再用一个按值传的 dup 记"路上重复出现了几次"：进门后 freq[val] ≥ 2 就 dup + 1，到叶子时 dup == 0 才计数。freq 按引用传必须手动撤销，dup 按值传自动恢复，两种恢复方式在同一个函数里对比着看。practice 里用"先收集所有路径、再逐条检查有无重复"的暴力解对拍。

参考答案
void distinctDfs(TreeNode* node, vector<int>& freq, int dup, int& cnt) {
    if (node == nullptr) return;
    freq[node->val]++;                           // 进门：这个编号在路上多出现一次
    if (freq[node->val] >= 2) dup++;             // 路上已经有过它：多一次重复（dup 按值传）
    if (node->left == nullptr && node->right == nullptr) {
        if (dup == 0) cnt++;
    } else {
        distinctDfs(node->left, freq, dup, cnt);
        distinctDfs(node->right, freq, dup, cnt);
    }
    freq[node->val]--;                           // 出门：撤销，和 path.pop_back() 同一个道理
}
int distinctPathCount(TreeNode* root) {
    vector<int> freq(10, 0);
    int cnt = 0;
    distinctDfs(root, freq, 0, cnt);
    return cnt;
}

    freq[node.val] += 1                          # 进门：这个编号在路上多出现一次
    if freq[node.val] >= 2: dup += 1             # 路上已经有过它：多一次重复（只改本层）
    if node.left is None and node.right is None:
        if dup == 0: cnt[0] += 1
    else:
        distinct_dfs(node.left, freq, dup, cnt)
        distinct_dfs(node.right, freq, dup, cnt)
    freq[node.val] -= 1                          # 出门：撤销，和 path.pop() 同一个道理
def distinct_path_count(root):

    cnt = [0]

    return cnt[0]

易错点

  • 在空节点处记录路径，每条路径会被记两次。
  • res.push_back(path) 之后接着 return，却把 pop_back 放在 return 之后：叶子的值永远不会被撤销。用 if/else 或者在 return 前 pop。
  • 把 sum 也按引用传却没有手动减回去，兄弟子树会继承错误的和。

小结： path 记录走过的路径：进入节点时 push，离开时 pop；到达叶子时复制一份存入结果。

6.8 从树到图：节点、边、邻接表和 visited 必读

社交关系、城市道路、课程依赖都不是树：一个点可以与许多点相连，沿着边走可能绕回原处，有些点之间根本不连通。这样的结构叫作图（graph）。把树上的 DFS、BFS 用到图上，只需补充两点：怎样存储图，以及怎样避免重复访问。

图由节点和边组成

图由节点（也叫顶点，vertex）和边（edge）组成，每条边连接两个节点。边没有方向的叫无向图（如朋友关系），有方向的叫有向图（如关注关系、课程的先修关系）。与树相比，图有两种新的情况：

  • 可能有环（cycle）：从一个点出发，沿着边能够走回自己。
  • 可能不连通：有些点之间没有任何路径。

树实际上是连通且没有环的特殊的图，所以在树上从根向下走永远不会回头，而在图上必须自己防止重复访问。

6 个点、5 条无向边。1→2→4→3→1 能绕回起点（有环）；5、6 和其他点之间没有边（不连通）

邻接表：每个点记录它的邻居

题目通常给出边列表：点数 n 和若干对 (u, v)。把它转换成邻接表（adjacency list）：adj[u] 是一个 vector，列出所有与 u 直接相连的点。点的编号从 1 开始时，开 n+1 个位置，下标 0 空着不用，这样 adj[u] 直接对应点 u。[1]

vector<vector<int>> buildGraph(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> adj(n + 1);          // 点编号 1..n，下标 0 不用
    for (auto [u, v] : edges) {
        adj[u].push_back(v);                 // 无向边：两头都记
        adj[v].push_back(u);
    }
    return adj;
}
def build_graph(n, edges):
    adj = [[] for _ in range(n + 1)]         # 点编号 1..n，下标 0 不用；写成 [[]] * (n + 1) 会让所有位置共用同一个 list
    for u, v in edges:
        adj[u].append(v)                     # 无向边：两头都记
        adj[v].append(u)
    return adj

void dfsNoVisited(const vector<vector<int>>& adj, int u, vector<int>& order, int limit) {
    if ((int)order.size() >= limit) return;  // 真代码里没有这一行，会一直递归到栈溢出
    order.push_back(u);
    for (int v : adj[u]) dfsNoVisited(adj, v, order, limit);
}
def dfs_no_visited(adj, u, order, limit):
    if len(order) >= limit: return       # 真代码里没有这一行；Python 默认约 1000 层就抛 RecursionError
    order.append(u)
    for v in adj[u]: dfs_no_visited(adj, v, order, limit)

limit = 8 时输出 1 2 1 2 1 2 1 2。去掉上限后，调用栈越来越高，最终栈溢出（stack overflow），评测显示运行时错误；BFS 不加 visited 时，队列会无限增长，表现为超时或内存超限。

改正的方法是准备一个 vector<bool> visited(n + 1, false)：第一次到达某个点时把它标记为 true，以后再遇到它就直接跳过。visited 只回答一个问题：这个点以前是否访问过。它保证每个点最多处理一次，所以图的 DFS 和 BFS 的时间都是 O(n + m)，其中 n 是点数，m 是边数。

要点

  1. 图由节点和边组成，可能有环，可能不连通；树是连通且无环的图。
  2. 做题时用邻接表 vector<vector<int>> adj(n + 1) 存图；无向边在两个端点上都要 push。
  3. 图的搜索必须有 visited，否则遇到环就会无限循环。
  4. 要遍历不连通的图，需要在外层对每个点检查：没有访问过就从它出发搜索一次（见 6.9 的 countComponents）。

提示

"关注"有方向，是有向图。粉丝数就是入度：指向这个点的边数。练一练求的是出度（adj[u].size()），这里换成数每条边的终点：对每条边 (u, v) 让 indeg[v] 加 1，连邻接表都不用建。找最大值时用严格大于，并列时就会保留编号小的。

int mostFollowed(int n, const vector<pair<int, int>>& edges) {
    vector<int> indeg(n + 1, 0);              // 下标 0 不用
    for (auto [u, v] : edges) indeg[v]++;     // 边 u -> v：v 多一个粉丝
    int best = 1;
    for (int u = 2; u <= n; ++u)
        if (indeg[u] > indeg[best]) best = u; // 严格大于：并列时保留编号小的
    return best;
}

vector<int> commonFriends(const vector<vector<int>>& adj, int a, int b) {

    vector<int> res;
    for (int v : adj[b])
        if (isFriendOfA[v]) res.push_back(v);
    sort(res.begin(), res.end());
    return res;
}

练习 6.8-3【综合】脏数据建图：去掉自环和重复边

题面：题目给的边列表可能有脏数据：自环 (u, u)、同一条边出现多次、(u, v) 和 (v, u) 同时出现。写 vector<vector<int>> buildSimpleGraph(int n, const vector<pair<int, int>>& edges)，建无向图的邻接表，要求丢掉自环，每对相连的点只记一次，并且每个 adj[u] 按编号从小到大。n ≤ 10⁵，边数不超过 2×10⁵。

样例：n = 3，边 (1,2) (2,1) (2,2) (1,3) (1,2) → adj[1] = {2, 3}，adj[2] = {1}，adj[3] = {1}。n = 2，没有边 → adj[1]、adj[2] 都为空。

提示

先照常建图（无向边两头都记），遇到 u == v 直接跳过。重复边会让同一个邻居在 adj[u] 里出现多次；对每个 adj[u] 排序后，相同的邻居挨在一起，再用 unique + erase 删掉相邻的重复。(u, v) 和 (v, u) 在两头记完之后也变成"同一个邻居出现两次"，同样会被去掉。重复边不处理时，6.9 的判环会把 1-2-1 误认为一个环，所以这一步在实际题目里很重要。practice 用 n×n 的布尔矩阵做暴力对拍。

参考答案
vector<vector<int>> buildSimpleGraph(int n, const vector<pair<int, int>>& edges) {
    vector<vector<int>> adj(n + 1);
    for (auto [u, v] : edges) {
        if (u == v) continue;                 // 自环：自己连自己，丢掉
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    for (int u = 1; u <= n; ++u) {
        sort(adj[u].begin(), adj[u].end());
        adj[u].erase(unique(adj[u].begin(), adj[u].end()), adj[u].end());   // 排序后相邻的重复只留一个
    }
    return adj;
}
def build_simple_graph(n, edges):
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        if u == v: continue                   # 自环：自己连自己，丢掉
        adj[u].append(v)
        adj[v].append(u)
    for u in range(1, n + 1):
        adj[u] = sorted(set(adj[u]))          # set 去重再排序，对应 sort + unique + erase
    return adj

易错点

  • 点编号从 1 开始，却开了 adj(n)，访问 adj[n] 越界。
  • 无向图只 push 了一个方向，从另一头出发的搜索会漏点。
  • 把树的 DFS 原样用在图上，没有 visited。

小结： 图由点和边组成，可能有环，也可能不连通；邻接表写成 adj(n+1)，无向边在两端各记录一次；visited 记录每个点是否已经访问过。
