12.15 最终安全状态：把图反过来再做拓扑

原题回顾

Kahn 的核心判断是"一个点的所有前驱都处理完了，它就可以处理"，用入度计数。

变化点

LeetCode 802：有向图里，没有出边的点叫终点。一个点如果从它出发的每一条路径最后都会停在终点（绝不会走进环里绕圈），就叫安全点。按编号从小到大输出所有安全点。[1]

为什么这样改：反向建图，出度当入度

"u 安全"等于"u 的所有后继都安全"。这和 Kahn 的"所有前驱都处理完"是镜像关系：把每条边掉个头，后继就变成了前驱。于是：

  • 位置 ①：反向建图 rg，同时统计原图的出度 outdeg[u]（u 还有几条出边没确认安全）。
  • 位置 ③：起点是出度为 0 的终点，它们一定安全。
  • 位置 ⑤：v 被确认安全后，沿反向边通知每个指向 v 的 u：outdeg[u] 减 1；减到 0 说明 u 的出边全部通向安全点，u 也安全。

环上的点出边永远有一条指向环内，出度减不到 0；能走进环的点也一样。所以最后没被确认的点就是不安全的。

例图：3 是终点（没有出边）；4、5 互相指，是一个环。从 2 出发可能走进环，所以 2 不安全；0 能走到 2，也不安全

代码

vector<int> safeNodes(const vector<vector<int>>& graph) {              // graph[u]：u 能走到的点
    int n = graph.size();
    vector<vector<int>> rg(n);
    vector<int> outdeg(n, 0);
    for (int u = 0; u < n; ++u)
        for (int v : graph[u]) { rg[v].push_back(u); ++outdeg[u]; }   // 反向：v 记住谁指向它
    queue<int> q;
    for (int u = 0; u < n; ++u) if (outdeg[u] == 0) q.push(u);         // 终点：一定安全
    vector<bool> safe(n, false);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        safe[v] = true;
        for (int u : rg[v]) if (--outdeg[u] == 0) q.push(u);           // u 的出边全通向安全点
    }
    vector<int> ans;
    for (int u = 0; u < n; ++u) if (safe[u]) ans.push_back(u);
    return ans;
}
def safe_nodes(graph):                                  # graph[u]：u 能走到的点
    n = len(graph)
    rg = [[] for _ in range(n)]
    outdeg = [0] * n
    for u in range(n):
        for v in graph[u]:                              # 反向：v 记住谁指向它
            rg[v].append(u)
            outdeg[u] += 1
    q = deque(u for u in range(n) if outdeg[u] == 0)    # 终点：一定安全
    safe = [False] * n
    while q:
        v = q.popleft()
        safe[v] = True
        for u in rg[v]:
            outdeg[u] -= 1
            if outdeg[u] == 0:                          # u 的出边全通向安全点
                q.append(u)
    return [u for u in range(n) if safe[u]]

逐行说明：

行作用为什么
rg[v].push_back(u); ++outdeg[u];反向存边，正向数出度确认 v 后要找"谁指向 v"
if (outdeg[u] == 0) q.push(u);终点入队没有出边，不可能走进环
if (--outdeg[u] == 0) q.push(u);u 的最后一条出边也确认了"所有后继都安全"才安全
最后按编号收集题目要求升序出队顺序不是编号顺序

10.12 的 DFS 三色法也能做：一个点在 DFS 中被染成黑色（后代全部查完、没碰到灰点），它就是安全的。两种方法都是 O(n + e)。

练一练：至少有一条路能到终点

题面： 把"每一条路"改成"至少一条路"：哪些点存在一条通向终点的路径？同一张图上，答案是 0 1 2 3 6，比安全点多了 0 和 2。

提示： "至少一条"不需要计数：沿反向边能走到的点全部标记即可，这就是反向图上从所有终点出发的多源 BFS。

vector<int> canReachTerminal(const vector<vector<int>>& graph) {       // 至少有一条路能到终点
    int n = graph.size();
    vector<vector<int>> rg(n);
    for (int u = 0; u < n; ++u)
        for (int v : graph[u]) rg[v].push_back(u);
    queue<int> q;
    vector<bool> ok(n, false);
    for (int u = 0; u < n; ++u)
        if (graph[u].empty()) { ok[u] = true; q.push(u); }
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (int u : rg[v])
            if (!ok[u]) { ok[u] = true; q.push(u); }                   // 改动：一条边通就够，不用计数
    }
    vector<int> ans;
    for (int u = 0; u < n; ++u) if (ok[u]) ans.push_back(u);

}
def can_reach_terminal(graph):                          # 至少有一条路能到终点
    n = len(graph)
    rg = [[] for _ in range(n)]
    for u in range(n):
        for v in graph[u]:
            rg[v].append(u)
    ok = [not graph[u] for u in range(n)]               # 终点先标记
    q = deque(u for u in range(n) if ok[u])
    while q:
        v = q.popleft()
        for u in rg[v]:
            if not ok[u]:                               # 改动：一条边通就够，不用计数
                ok[u] = True
                q.append(u)
    return [u for u in range(n) if ok[u]]

练习 12.15-1【基础】一定会停机的状态

题面：一台状态机有 n 个状态，graph[u] 列出状态 u 可能跳转到的状态（可以跳回自己）；没有任何跳转的状态是停机状态。从某个状态出发，如果不论怎么跳最后都会停机，就称它"必停"。写 vector<int> haltingStates(graph)，按编号升序返回所有必停状态。

样例：graph = [[1],[2],[],[3,4],[2]]，输出 [0,1,2,4]。2 是停机状态；0、1、4 都只能跳向必停状态；3 可以跳回自己，可能永远停不下来。graph = [[0]]，输出 []。

提示

状态是点，跳转是边，停机状态是没有出边的终点，"不论怎么跳都会停机"就是"每条路径都走到终点"，即 12.15 的安全点。做法沿用 safeNodes：反向建图，出度当入度，终点做起点。自环 3→3 让 3 的出度永远减不到 0，3 不会被确认，这正好对应"可能一直停在 3 打转"。

参考答案
vector<int> haltingStates(const vector<vector<int>>& graph) {
    int n = graph.size();
    vector<vector<int>> rg(n);
    vector<int> outdeg(n, 0);
    for (int u = 0; u < n; ++u)
        for (int v : graph[u]) { rg[v].push_back(u); ++outdeg[u]; }   // 自环也算一条出边
    queue<int> q;
    for (int u = 0; u < n; ++u) if (outdeg[u] == 0) q.push(u);         // 停机状态
    vector<bool> ok(n, false);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        ok[v] = true;
        for (int u : rg[v]) if (--outdeg[u] == 0) q.push(u);
    }
    vector<int> ans;
    for (int u = 0; u < n; ++u) if (ok[u]) ans.push_back(u);
    return ans;
}
def halting_states(graph):

练习 12.15-2【变形】只有到达目标才算成功

题面：有向图 graph 和标记数组 goal。路径走到任何一个 goal 点就算成功并立即停止（goal 点的出边不再走）；走到一个没有出边的非 goal 点算失败（死胡同）；永远走不完也算失败。一个点如果从它出发的每条路径都成功，就称它"稳赢"。写 vector<int> alwaysReachGoal(graph, goal)，按编号升序返回所有稳赢的点。

样例：graph = [[1,2],[3],[],[1],[3]]，goal = [false,false,false,true,false]，输出 [1,3,4]。2 没有出边又不是目标，是死胡同；0 可能走到 2，不稳赢；3 自己是目标，它的出边 3→1 不算数。

提示

和安全点比有两处不同。起点不能沿用 safeNodes 的"出度为 0"：出度为 0 的非 goal 点是死胡同，有出边的 goal 点反而是成功的终点，所以起点按 goal 标记放。死胡同不入队，它和所有可能走到它的点就永远不会被确认。goal 点的出边在建图时整条跳过：路径到了 goal 就停，这些边并不存在；保留它们的话，goal 点的后继全被确认时它的出度减到 0，会第二次入队。其余沿用 safeNodes：一个点的出边全部指向已确认的点，它就被确认。

参考答案
vector<int> alwaysReachGoal(const vector<vector<int>>& graph, const vector<bool>& goal) {
    int n = graph.size();
    vector<vector<int>> rg(n);
    vector<int> outdeg(n, 0);
    for (int u = 0; u < n; ++u) {
        if (goal[u]) continue;                               // 到了目标就停：它的出边不算
        for (int v : graph[u]) { rg[v].push_back(u); ++outdeg[u]; }
    }
    queue<int> q;
    for (int u = 0; u < n; ++u) if (goal[u]) q.push(u);     // 起点只放目标，死胡同不入队
    vector<bool> ok(n, false);
    while (!q.empty()) {
        int v = q.front(); q.pop();
        ok[v] = true;
        for (int u : rg[v]) if (--outdeg[u] == 0) q.push(u);
    }
    vector<int> ans;
    for (int u = 0; u < n; ++u) if (ok[u]) ans.push_back(u);
    return ans;
}
def always_reach_goal(graph, goal):
    n = len(graph)
    rg = [[] for _ in range(n)]
    outdeg = [0] * n
    for u in range(n):
        if goal[u]:                                     # 到了目标就停：它的出边不算
            continue
        for v in graph[u]:
            rg[v].append(u)
            outdeg[u] += 1
    q = deque(u for u in range(n) if goal[u])           # 起点只放目标，死胡同不入队
    ok = [False] * n
    while q:
        v = q.popleft()
        ok[v] = True
        for u in rg[v]:
            outdeg[u] -= 1
            if outdeg[u] == 0:
                q.append(u)
    return [u for u in range(n) if ok[u]]

练习 12.15-3【综合】安全的点最多还要走几步

题面：有向图 graph，终点是没有出边的点。对每个安全点（每条路径都走到终点），求从它出发最多走几步就一定停下，也就是到终点的最长路径的边数；不安全的点记 −1。写 vector<int> maxStepsToStop(graph)。

样例：graph = [[1,2],[2],[],[3]]，输出 [2,1,0,−1]：2 是终点；1 走一步到 2；0 可以直接到 2，也可以经过 1 再到 2，最多 2 步；3 有自环，不安全。graph = [[1],[0]]，输出 [−1,−1]。

提示

安全点出发的路径都有限，最长那条满足 steps[u] = 1 + max(steps[v])，v 取遍 u 的后继，终点为 0。要算 steps[u]，必须先算完 u 的全部后继；反向 Kahn 里 u 入队的时刻，正是它的出边全部指向已确认的点的时刻，所以确认 v 时沿反向边顺手更新 steps[u] = max(steps[u], steps[v] + 1)。这和 12.12 的"等最慢的前驱"是同一个结构，只是方向反了过来。没被确认的点保持 −1。

参考答案
vector<int> maxStepsToStop(const vector<vector<int>>& graph) {
    int n = graph.size();
    vector<vector<int>> rg(n);
    vector<int> outdeg(n, 0);
    for (int u = 0; u < n; ++u)
        for (int v : graph[u]) { rg[v].push_back(u); ++outdeg[u]; }
    vector<int> steps(n, -1);
    queue<int> q;
    for (int u = 0; u < n; ++u) if (outdeg[u] == 0) { steps[u] = 0; q.push(u); }   // 终点：0 步
    while (!q.empty()) {
        int v = q.front(); q.pop();                          // 出队时 steps[v] 已经确定
        for (int u : rg[v]) {
            steps[u] = max(steps[u], steps[v] + 1);          // 取最慢的那个后继
            if (--outdeg[u] == 0) q.push(u);
        }
    }
    for (int u = 0; u < n; ++u) if (outdeg[u] > 0) steps[u] = -1;   // 没被确认：不安全
    return steps;
}
def max_steps_to_stop(graph):
    n = len(graph)
    rg = [[] for _ in range(n)]
    outdeg = [0] * n
    for u in range(n):
        for v in graph[u]:
            rg[v].append(u)
            outdeg[u] += 1
    steps = [-1] * n
    q = deque()
    for u in range(n):
        if outdeg[u] == 0:                              # 终点：0 步
            steps[u] = 0
            q.append(u)
    while q:
        v = q.popleft()                                 # 出队时 steps[v] 已经确定
        for u in rg[v]:
            steps[u] = max(steps[u], steps[v] + 1)      # 取最慢的那个后继
            outdeg[u] -= 1
            if outdeg[u] == 0:
                q.append(u)
    return [steps[u] if outdeg[u] == 0 else -1 for u in range(n)]   # 没被确认：不安全

易错点：

  • 正向图上从终点出发做 BFS：边的方向反了，走不到任何点。
  • 把"能到终点"当成"安全"：2 能走到终点 3，也能走进环 4⇄5，所以不安全。
  • 输出时按出队顺序：题目要升序，要按编号重新收集。

小结： 求安全点时建反向图，把原图的出度当作入度，从终点出发；要求所有后继都满足条件时用计数，只要求存在一条时用标记。

12.16 最小高度树：在无向树上一圈圈剥叶子

原题回顾

拓扑排序处理的是有向图，起点是入度为 0 的点，每一轮把它们"删掉"，露出新的起点。

变化点

LeetCode 310：给一棵 n 个点的无向树，任选一个点当根，树高是根到最远叶子的边数。求所有能使树高最小的根。[2]

为什么这样改：从叶子往里剥

最直接的做法是每个点都做一次 BFS 求树高，O(n²)。更快的想法是：最好的根应该在"正中间"，离各个方向的叶子都尽量近。把所有叶子同时剥掉一圈，树的中心不变；一圈圈剥下去，最后剩下的 1 个或 2 个点就是中心，也就是最小高度树的根。可以证明答案最多 2 个：它们是树上最长路径（直径）的中点，直径边数为偶数时中点是 1 个点，为奇数时是中间那条边的 2 个端点。

套进骨架：

  • 位置 ①：无向图，存每个点的度（相连的边数）。
  • 位置 ③：起点是度为 1 的叶子。
  • 位置 ②④：按层出队，一层就是一圈叶子。
  • 位置 ⑤：邻居的度减 1，减到 1 就成了新叶子。
  • 位置 ⑥：剩下的点数 remain ≤ 2 时停，队列里的就是答案。

代码

vector<int> findMinHeightTrees(int n, const vector<pair<int, int>>& edges) {
    if (n == 1) return {0};
    vector<vector<int>> g(n);
    vector<int> deg(n, 0);
    for (auto [a, b] : edges) { g[a].push_back(b); g[b].push_back(a); ++deg[a]; ++deg[b]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (deg[i] == 1) q.push(i);           // 叶子：度为 1
    int remain = n;
    while (remain > 2) {
        int cnt = q.size();
        remain -= cnt;                                                 // 这一圈叶子全部剥掉
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            for (int v : g[u]) if (--deg[v] == 1) q.push(v);           // 邻居变成新叶子
        }
    }
    vector<int> ans;
    while (!q.empty()) { ans.push_back(q.front()); q.pop(); }
    sort(ans.begin(), ans.end());
    return ans;
}

练习 12.16-3【综合】先确认是一棵树

题面：给 n 个点（编号 0..n−1，n ≥ 1）和一组无向边 edges，边数不定，没有重边和自环。如果它构成一棵树，写 vector<int> centersIfTree(int n, edges) 返回最小高度树的根（升序）；不是树就返回空数组。

样例：n = 4，edges = [(0,1),(1,2),(1,3)]，输出 [1]。n = 4，edges = [(0,1),(1,2),(2,0)]，输出 []：边数恰好是 n − 1，但 0、1、2 成环，3 孤立。n = 3，edges = [(0,1)]，输出 []。n = 1 没有边，输出 [0]。

提示

n 个点是树，当且仅当恰好 n − 1 条边并且连通，两个条件都要查：第二个样例边数对，却不连通。不先检查就直接剥叶子会出事：环上的点度数始终 ≥ 2，孤立点度数是 0，它们都不会入队；队列空了而 remain 仍大于 2，while (remain > 2) 永远不结束。所以先数边数，再从 0 号做一次 BFS 看能否到达全部 n 个点，都通过后才照 findMinHeightTrees 剥叶子。

参考答案
vector<int> centersIfTree(int n, const vector<pair<int, int>>& edges) {
    if ((int)edges.size() != n - 1) return {};               // 树恰好 n-1 条边
    vector<vector<int>> g(n);
    vector<int> deg(n, 0);
    for (auto [a, b] : edges) { g[a].push_back(b); g[b].push_back(a); ++deg[a]; ++deg[b]; }
    vector<bool> vis(n, false);                              // BFS 检查连通
    queue<int> bq;
    bq.push(0);
    vis[0] = true;
    int seen = 1;
    while (!bq.empty()) {
        int u = bq.front(); bq.pop();
        for (int v : g[u]) if (!vis[v]) { vis[v] = true; ++seen; bq.push(v); }
    }
    if (seen < n) return {};                                 // 不连通：不是树，剥叶子会死循环
    if (n == 1) return {0};
    queue<int> q;
    for (int i = 0; i < n; ++i) if (deg[i] == 1) q.push(i);
    int remain = n;
    while (remain > 2) {
        int cnt = q.size();
        remain -= cnt;
        for (int i = 0; i < cnt; ++i) {
            int u = q.front(); q.pop();
            for (int v : g[u]) if (--deg[v] == 1) q.push(v);
        }
    }
    vector<int> ans;
    while (!q.empty()) { ans.push_back(q.front()); q.pop(); }
    sort(ans.begin(), ans.end());
    return ans;
}

        g[b].append(a)

        deg[b] += 1

                seen += 1
                bq.append(v)

  • 一个一个地剥、不按层：剩下的 2 个点可能不是中心。
  • 忘了 n == 1：唯一的点度为 0，队列一开始就是空的，返回空列表。

12.17 矩阵中的最长递增路径：网格里藏着一张 DAG

原题回顾

前面的图都直接给出点和边。

变化点

LeetCode 329：给一个整数矩阵，每步可以走到上下左右相邻的格子，要求每一步的值严格变大。求最长路径的格子数。[3]

为什么这样改：小的指向大的，就是一张 DAG

把每个格子看成一个点，从小格子向相邻的大格子连一条边。沿着边走值一直变大，不可能回到起点，所以没有环，这是一张隐式的 DAG（边没有存下来，需要时现算）。最长递增路径就是 DAG 上的最长路径。

求 DAG 最长路径正好用 12.11 的按层 Kahn：入度 = 四周比我小的邻居个数，入度为 0 的格子（四周没有更小的）是路径起点；层数就是最长路径的格子数，道理和"最少学期数 = 最长依赖链"一样。

代码一：按层拓扑

int longestIncreasingPath(const vector<vector<int>>& a) {
    int R = a.size(), C = a[0].size();
    const int dr[4] = {-1, 1, 0, 0}, dc[4] = {0, 0, -1, 1};
    vector<vector<int>> indeg(R, vector<int>(C, 0));                   // 比我小的邻居个数
    for (int r = 0; r < R; ++r)
        for (int c = 0; c < C; ++c)
            for (int d = 0; d < 4; ++d) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr >= 0 && nr < R && nc >= 0 && nc < C && a[nr][nc] < a[r][c]) ++indeg[r][c];
            }
    queue<pair<int, int>> q;
    for (int r = 0; r < R; ++r)
        for (int c = 0; c < C; ++c)
            if (indeg[r][c] == 0) q.push({r, c});                      // 四周没有更小的：路径起点
    int layers = 0;
    while (!q.empty()) {
        ++layers;
        int cnt = q.size();
        for (int i = 0; i < cnt; ++i) {
            auto [r, c] = q.front(); q.pop();
            for (int d = 0; d < 4; ++d) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr >= 0 && nr < R && nc >= 0 && nc < C && a[nr][nc] > a[r][c]
                    && --indeg[nr][nc] == 0) q.push({nr, nc});
            }
        }
    }
    return layers;                                                     // 层数 = 最长路径的格子数
}
def longest_increasing_path(a):
    R, C = len(a), len(a[0])
    dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    indeg = [[0] * C for _ in range(R)]                 # 比我小的邻居个数
    for r in range(R):
        for c in range(C):
            for dr, dc in dirs:
                nr, nc = r + dr, c + dc
                if 0 <= nr < R and 0 <= nc < C and a[nr][nc] < a[r][c]:
                    indeg[r][c] += 1
    q = deque((r, c) for r in range(R) for c in range(C) if indeg[r][c] == 0)   # 路径起点
    layers = 0
    while q:
        layers += 1
        for _ in range(len(q)):
            r, c = q.popleft()
            for dr, dc in dirs:
                nr, nc = r + dr, c + dc
                if 0 <= nr < R and 0 <= nc < C and a[nr][nc] > a[r][c]:
                    indeg[nr][nc] -= 1
                    if indeg[nr][nc] == 0:
                        q.append((nr, nc))
    return layers                                       # 层数 = 最长路径的格子数

代码二：记忆化 DFS

另一种常见写法：设 dfs(r,c) 为"从 (r,c) 出发的最长递增路径长度"，它等于 1 + 四周更大邻居的 dfs 的最大值。同一个格子会被问很多次，第一次算完存进 memo，以后直接用。因为图无环，递归一定会停。

int lipDfs(const vector<vector<int>>& a, vector<vector<int>>& memo, int r, int c) {
    if (memo[r][c]) return memo[r][c];                                 // 算过就直接用
    const int dr[4] = {-1, 1, 0, 0}, dc[4] = {0, 0, -1, 1};
    int best = 1;                                                      // 只有自己：长度 1
    for (int d = 0; d < 4; ++d) {
        int nr = r + dr[d], nc = c + dc[d];
        if (nr >= 0 && nr < (int)a.size() && nc >= 0 && nc < (int)a[0].size() && a[nr][nc] > a[r][c])
            best = max(best, 1 + lipDfs(a, memo, nr, nc));
    }
    return memo[r][c] = best;
}
int longestIncreasingPathMemo(const vector<vector<int>>& a) {
    vector<vector<int>> memo(a.size(), vector<int>(a[0].size(), 0));
    int ans = 0;
    for (int r = 0; r < (int)a.size(); ++r)
        for (int c = 0; c < (int)a[0].size(); ++c) ans = max(ans, lipDfs(a, memo, r, c));
    return ans;
}
def lip_dfs(a, memo, r, c):
    if memo[r][c]:                                      # 算过就直接用
        return memo[r][c]
    best = 1                                            # 只有自己：长度 1
    for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
        nr, nc = r + dr, c + dc
        if 0 <= nr < len(a) and 0 <= nc < len(a[0]) and a[nr][nc] > a[r][c]:
            best = max(best, 1 + lip_dfs(a, memo, nr, nc))
    memo[r][c] = best
    return best

def longest_increasing_path_memo(a):                    # 递归深度可达 R*C：大矩阵要调 sys.setrecursionlimit
    memo = [[0] * len(a[0]) for _ in range(len(a))]
    return max(lip_dfs(a, memo, r, c) for r in range(len(a)) for c in range(len(a[0])))

两种写法都是 O(R×C)。记忆化 DFS 更短；按层拓扑没有递归，矩阵很大时不用担心栈溢出。

练习 12.17-2【变形】一共有多少条递增路径

题面：给 R × C 的整数矩阵 a（R × C ≤ 10⁵），写 long long countIncreasingPaths(a)：每一步走到上下左右相邻且值严格更大的格子，单独一个格子也算一条路径。返回严格递增路径的条数对 10⁹ + 7 取模的结果。

样例：a = [[1,1],[3,4]]，输出 8：单格 4 条；两格的有 3 条，左上的 1 到 3、右上的 1 到 4、3 到 4；三格的只有 1、3、4 这一条。两个 1 相等，彼此之间不能走。a = [[1],[2]]，输出 3。a = [[5,5]]，输出 2。

提示

设 cnt[c] 为以格子 c 结尾的递增路径条数：只有它自己的 1 条，加上从每个更小邻居 p 走过来的 cnt[p] 条。算 cnt[c] 之前要先算完所有更小的邻居，拓扑序正好保证这一点；这里只关心顺序、不关心层数，所以不用分层。u 出队时 cnt[u] 已经完整，沿边把它加到每个更大的邻居上。答案是全部 cnt 之和。路径条数随格子数指数增长，每次加法后都对 10⁹ + 7 取模。

参考答案
long long countIncreasingPaths(const vector<vector<int>>& a) {
    const long long MOD = 1000000007;
    int R = a.size(), C = a[0].size();
    const int dr[4] = {-1, 1, 0, 0}, dc[4] = {0, 0, -1, 1};
    vector<vector<int>> indeg(R, vector<int>(C, 0));
    vector<vector<long long>> cnt(R, vector<long long>(C, 1));   // 只有自己：1 条
    for (int r = 0; r < R; ++r)
        for (int c = 0; c < C; ++c)
            for (int d = 0; d < 4; ++d) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr >= 0 && nr < R && nc >= 0 && nc < C && a[nr][nc] < a[r][c]) ++indeg[r][c];
            }
    queue<pair<int, int>> q;
    for (int r = 0; r < R; ++r)
        for (int c = 0; c < C; ++c)
            if (indeg[r][c] == 0) q.push({r, c});
    long long total = 0;
    while (!q.empty()) {
        auto [r, c] = q.front(); q.pop();                    // 出队时 cnt[r][c] 已经完整
        total = (total + cnt[r][c]) % MOD;
        for (int d = 0; d < 4; ++d) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < R && nc >= 0 && nc < C && a[nr][nc] > a[r][c]) {
                cnt[nr][nc] = (cnt[nr][nc] + cnt[r][c]) % MOD;   // 从 (r,c) 走过去的路径
                if (--indeg[nr][nc] == 0) q.push({nr, nc});
            }
        }
    }
    return total;
}

练习 12.17-3【综合】八个方向，还有障碍

题面：R × C 的矩阵 a，值为 −1 的格子是障碍，不能进入；其余格子的值都 ≥ 0。每一步可以走到周围 8 个方向（含斜向）相邻、非障碍、且值严格更大的格子。写 int longestIncreasing8(a) 返回最长递增路径的格子数；全是障碍时返回 0。

样例：a = [[1,−1,4],[2,3,−1],[−1,6,5]]，输出 5：1、2、3、5、6，其中 3 到 5 是斜着走的；若只能走 4 个方向，最长只有 1、2、3、6 共 4 格。a = [[−1,−1]] 输出 0。a = [[0,3],[2,1]] 输出 4：0、1、2、3，其中 0 到 1、2 到 3 都是斜着走；只走 4 个方向时最长只有 2 格。

提示

只改"谁是邻居"。方向数组扩成 8 个，多出四个斜向 (±1, ±1)。障碍格既不是点也不是邻居：统计入度时跳过以障碍为中心的格子，也跳过障碍邻居；起点只从非障碍格里找。沿边更新时同样跳过障碍。障碍的值是 −1，比所有正常值都小，如果忘了跳过，它会被当成局部最低点入队，还会给周围格子贡献入度，路径就能穿过障碍。全是障碍时没有起点，层数为 0。

易错点：

  • 相邻格子值相等时也连边：条件变成"不减"，相等的两格会互相指，出现环，按层拓扑的格子数会少算，记忆化 DFS 会无限递归。
  • 记忆化数组用 0 表示"没算过"：每个格子的答案至少是 1，所以 0 不会和真实答案混淆；如果答案可能是 0，就得用 −1 当标记。
  • Python 记忆化递归深度可能达到 R×C，大矩阵要调高 sys.setrecursionlimit 或改用按层拓扑。

小结： 相邻格子之间从小值指向大值，严格递增的要求使网格成为一个隐式的有向无环图；入度是更小邻居的个数，按层 Kahn 的层数就是答案；用记忆化 DFS 求解与之等价。

12.18 先修课查询：沿拓扑序传递可达集合

原题回顾

课程表 207/210 只问整体能不能修完、给一个顺序。

变化点

LeetCode 1462：n 门课（n ≤ 100）和先修关系，另有很多查询 (u, v)：u 是不是 v 的先修课？直接或间接都算：0 是 1 的先修、1 是 2 的先修，那么 0 也是 2 的先修。[4]

为什么这样改：在位置 ⑤ 顺手传递集合

每个查询都做一次 BFS 是 O(n + e)，查询多了就慢。我们先一次性算出每门课的全部先修集合 pre[v]，之后每个查询 O(1)。

v 的先修 = 它的每个直接先修 u，加上 u 的全部先修：

pre[v] = ∪ ( pre[u] ∪ {u} )，u 取遍 v 的直接先修

和 12.12 一样，要算 pre[v] 必须先把它所有直接先修的 pre 算完，按拓扑序处理正好满足：u 出队时 pre[u] 已经完整，沿边把它并进 pre[v]。

集合用什么存？n ≤ 100，可以用 bitset<100>：第 u 位是 1 表示 u 在集合里，两个集合求并就是按位或 |=，100 位只要几次机器运算。[5] Python 的整数位数不限，直接拿 int 当位集合用。

代码

vector<bool> checkIfPrerequisite(int n, const vector<pair<int, int>>& edges,
                                 const vector<pair<int, int>>& queries) {   // n ≤ 100
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    vector<bitset<100>> pre(n);                                        // pre[v] 第 u 位为 1：u 是 v 的先修
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop();                                    // 此刻 pre[u] 已经完整
        for (int v : g[u]) {
            pre[v] |= pre[u];                                          // u 的先修也是 v 的先修
            pre[v][u] = 1;                                             // u 自己也是
            if (--indeg[v] == 0) q.push(v);
        }
    }
    vector<bool> ans;
    for (auto [u, v] : queries) ans.push_back(pre[v][u]);
    return ans;
}
def check_if_prerequisite(n, edges, queries):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    pre = [0] * n                                       # 用整数当位集合：Python 的 int 位数不限
    q = deque(i for i in range(n) if indeg[i] == 0)
    while q:
        u = q.popleft()                                 # 此刻 pre[u] 已经完整
        for v in g[u]:
            pre[v] |= pre[u] | (1 << u)                 # u 的先修和 u 自己都是 v 的先修
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return [(pre[v] >> u) & 1 == 1 for u, v in queries]

练习 12.18-1【基础】谁是谁的上级

题面：公司有 n 名员工（n ≤ 100，编号 0..n−1），reports 中的 (boss, staff) 表示 boss 是 staff 的直接上级，汇报关系不成环。queries 中每个 (a, b) 问：a 是不是 b 的上级（直接或间接都算）？写 vector<bool> isAboveQueries(int n, reports, queries)。

样例：n = 5，reports = [(0,1),(0,2),(1,3),(3,4)]，queries = [(0,4),(2,4),(1,4),(4,1)]，输出 [true,false,true,false]。0 经 1、3 管到 4；2 和 4 不在一条汇报链上；4 是 1 的下级而不是上级。

提示

上级关系就是先修关系：边从 boss 指向 staff，"a 是 b 的上级"就是"a 是 b 的祖先"。沿用 checkIfPrerequisite：沿拓扑序传递 above[v] |= above[u]，再把 u 自己加进去，查询看 above[b] 的第 a 位。

参考答案
vector<bool> isAboveQueries(int n, const vector<pair<int, int>>& reports,
                            const vector<pair<int, int>>& queries) {   // n ≤ 100
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [boss, staff] : reports) { g[boss].push_back(staff); ++indeg[staff]; }
    vector<bitset<100>> above(n);                            // above[v]：v 的全部上级
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) {
            above[v] |= above[u];                            // u 的上级也是 v 的上级
            above[v][u] = 1;                                 // u 自己是 v 的直接上级
            if (--indeg[v] == 0) q.push(v);
        }
    }
    vector<bool> ans;
    for (auto [a, b] : queries) ans.push_back(above[b][a]);
    return ans;
}
def is_above_queries(n, reports, queries):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for boss, staff in reports:
        g[boss].append(staff)
        indeg[staff] += 1
    above = [0] * n                                     # above[v]：v 的全部上级（整数当位集合）
    q = deque(i for i in range(n) if indeg[i] == 0)
    while q:
        u = q.popleft()
        for v in g[u]:
            above[v] |= above[u] | (1 << u)             # u 的上级和 u 自己都是 v 的上级
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return [(above[b] >> a) & 1 == 1 for a, b in queries]

练习 12.18-2【变形】每门课是多少门课的先修

题面：n 门课（n ≤ 100）、先修边表 edges，保证无环。写 vector<int> countDependents(int n, edges)：对每门课 u，返回有多少门课把 u 当作先修（直接或间接）。

样例：12.18 手工模拟的 5 门课（边 0→1、1→2、3→2、2→4），输出 [3,2,1,2,0]：0 是 1、2、4 的先修；3 是 2、4 的先修；4 不是任何课的先修。

提示

方向反过来：u 的"后修集合" post[u] 是它每个直接后继 v 的 post[v] 再加上 v 本身。算 post[u] 时要先算完它的所有后继，所以处理顺序是拓扑序倒过来：先用 Kahn 求出拓扑序，再从后往前遍历，post[u] |= post[v]，post[u][v] = 1。和 countAllPrereqs 相比，传递的方向和遍历的方向都反了。

参考答案
vector<int> countDependents(int n, const vector<pair<int, int>>& edges) {   // n ≤ 100
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    vector<int> order;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : g[u]) if (--indeg[v] == 0) q.push(v);
    }
    vector<bitset<100>> post(n);                             // post[u]：以 u 为先修的课
    for (int i = n - 1; i >= 0; --i) {                       // 倒着走：后继先算完
        int u = order[i];
        for (int v : g[u]) {
            post[u] |= post[v];
            post[u][v] = 1;
        }
    }
    vector<int> cnt(n);
    for (int u = 0; u < n; ++u) cnt[u] = post[u].count();
    return cnt;
}
def count_dependents(n, edges):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    post = [0] * n                                      # post[u]：以 u 为先修的课
    for u in reversed(order):                           # 倒着走：后继先算完
        for v in g[u]:
            post[u] |= post[v] | (1 << v)
    return [bin(p).count("1") for p in post]

练习 12.18-3【综合】两门课的共同先修，录入可能有环

题面：n 门课（n ≤ 100）、先修边表 edges，录入可能出错而形成环。queries 中每个 (a, b) 问：有多少门课同时是 a 和 b 的先修（直接或间接）？写 vector<int> commonPrereqCount(int n, edges, queries)；如果先修关系有环，所有查询都返回 −1。

样例：12.18 手工模拟的 5 门课，queries = [(2,4),(1,3),(4,4)]，输出 [3,0,4]：2 的先修是 {0,1,3}，4 的先修是 {0,1,2,3}，交集 3 门；3 没有先修；a = b 时就是它自己的先修门数。在边表里再加一条 (4,0)，输出 [−1,−1,−1]。

提示

pre 照 checkIfPrerequisite 算，两个集合的交集是按位与，个数用 count()。有环时 Kahn 会提前停下，环上和环下游的课 pre 都不完整，查询结果没有意义，所以先数出队个数，少于 n 就全部返回 −1。Python 里交集是 pre[a] & pre[b]，个数用 bin(x).count("1")。

参考答案
vector<int> commonPrereqCount(int n, const vector<pair<int, int>>& edges,
                              const vector<pair<int, int>>& queries) {   // n ≤ 100
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);
    for (auto [u, v] : edges) { g[u].push_back(v); ++indeg[v]; }
    vector<bitset<100>> pre(n);
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indeg[i] == 0) q.push(i);
    int done = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++done;
        for (int v : g[u]) {
            pre[v] |= pre[u];
            pre[v][u] = 1;
            if (--indeg[v] == 0) q.push(v);
        }
    }
    vector<int> ans;
    for (auto [a, b] : queries)
        ans.push_back(done < n ? -1 : (int)(pre[a] & pre[b]).count());   // 有环：pre 不完整
    return ans;
}
def common_prereq_count(n, edges, queries):
    g = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        g[u].append(v)
        indeg[v] += 1
    pre = [0] * n
    q = deque(i for i in range(n) if indeg[i] == 0)
    done = 0
    while q:
        u = q.popleft()
        done += 1
        for v in g[u]:
            pre[v] |= pre[u] | (1 << u)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return [-1 if done < n else bin(pre[a] & pre[b]).count("1") for a, b in queries]   # 有环：pre 不完整

易错点：

  • 不按拓扑序、按编号顺序传递：处理 v 时它的某个先修还没算完，间接关系会漏。
  • 只写了 pre[v] |= pre[u]，忘了 pre[v][u] = 1：直接先修丢失，只剩间接的。
  • 查询方向弄反：问"u 是不是 v 的先修"，要看 u 在不在 pre[v] 里，而不是 v 在不在 pre[u] 里。

小结： 可达性查询沿拓扑序传播：pre[v] |= pre[u] | {u}；n 较小时用 bitset，Python 中可以用 int 作为位集合；每次查询 O(1)。
